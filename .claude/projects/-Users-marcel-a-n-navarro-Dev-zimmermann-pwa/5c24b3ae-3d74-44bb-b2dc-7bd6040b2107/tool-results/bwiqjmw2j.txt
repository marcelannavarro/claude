   139	/**
   140	 * Contentful's `include`-resolved response nests navGroups/links/regions but applies no
   141	 * region scoping of its own — narrow each region-aware array down to the current
   142	 * storefront via the shared resolveByRegion utilities.
   143	 */
   144	export const scopeToRegion = (config: FooterConfigEntry, siteId: string): FooterConfigEntry => {
   145	  const navGroups: FooterNavGroupEntry[] = config.fields.navGroups.map(
   146	    (group): FooterNavGroupEntry => ({
   147	      ...group,
   148	      fields: { ...group.fields, links: filterByRegion<LinkEntry>(group.fields.links, siteId) },
   149	    }),
   150	  );
   151	  const legalLinks: LinkEntry[] | undefined = config.fields.legalLinks
   152	    ? filterByRegion<LinkEntry>(config.fields.legalLinks, siteId)
   153	    : undefined;
   154	
   155	  return { ...config, fields: { ...config.fields, navGroups, legalLinks } };
   156	};

   158	/**
   159	 * The component SFCC actually renders. Base `_app/index.jsx` resolves
   160	 * `@salesforce/retail-react-app/app/components/footer` through the overrides system and renders
   161	 * `<Footer />` with **no props**, so there is no upstream container to own the data. This
   162	 * wrapper is that seam, and the only place the Contentful fetch lives.
   163	 *
   164	 * HPC-034: fetches `footerConfig` entries via the Delivery API, then picks the entry scoped
   165	 * to the current storefront's region (falls back to the global/unscoped entry once one
   166	 * exists). Renders nothing while loading, on fetch failure, or when no entry resolves for
   167	 * this region — no fallback to placeholder content once wired to live data, so a Contentful
   168	 * outage doesn't silently ship stale placeholder copy to real shoppers. Note this puts a
   169	 * fetch inside `components/`, which components.md reserves for containers; if that matters,
   170	 * this should move into an `_app` override instead (Tier 2 shared shell — coordinate first).
   171	 */
   172	const FooterContainer = (props: Omit<FooterProps, 'data'>) => {
   173	  // buildUrl + history.push, not useNavigation. useNavigation special-cases the home path
   174	  // (`path === '/' ? '/' : updatedHref`), so navigate('/') pushes a bare '/' and skips the
   175	  // site/locale prefix — on a prefixed storefront that drops the shopper onto the default
   176	  // site. This is the exact pair base _app's onLogoClick uses, so both wordmarks resolve home
   177	  // the same way on every storefront.
   178	  //
   179	  // useMultiSite ships untyped from the base SFCC package. site.id here is the resolved
   180	  // SFCC site ID — confirmed against useCountryConfig.ts's identical client-side pattern
   181	  // (`c.sfccSiteId === site?.id`), the existing precedent for this exact lookup.
   182	  const { site, buildUrl } = useMultiSite() as {
   183	    site: { id: string };
   184	    buildUrl: (path: string, siteRef?: string, localeRef?: string) => string;
   185	  };
   186	  const history = useHistory();
   187	  const { client } = useContentfulClient();
   188	  const [footerData, setFooterData] = useState<FooterConfigEntry | null>(null);
   189	
   190	  useEffect(() => {
   191	    if (!client) return;
   192	    let cancelled = false;
   193	
   194	    client
   195	      .getEntries({ content_type: 'footerConfig', include: FOOTER_INCLUDE_DEPTH })
   196	      .then((response) => {
   197	        if (cancelled) return;
   198	        // The contentful SDK client is loosely typed (see useContentfulClient.ts) —
   199	        // FooterConfigEntry mirrors the exact nested-resolved shape `include: 3` produces
   200	        // for this content model.
   201	        const configs = response.items as unknown as FooterConfigEntry[];
   202	        const resolved = pickMostSpecificByRegion<FooterConfigEntry>(configs, site.id);
   203	        setFooterData(resolved ? scopeToRegion(resolved, site.id) : null);
   204	      })
   205	      .catch(() => {
   206	        if (!cancelled) setFooterData(null);
   207	      });
   208	
   209	    return () => {
   210	      cancelled = true;
   211	    };
   212	  }, [client, site.id]);
   213	
   214	  if (!footerData) return null;
   215	
   216	  return <Footer data={footerData} onLogoClick={() => history.push(buildUrl(HOME_HREF))} {...props} />;
   217	};
   218	
   219	export default FooterContainer;
