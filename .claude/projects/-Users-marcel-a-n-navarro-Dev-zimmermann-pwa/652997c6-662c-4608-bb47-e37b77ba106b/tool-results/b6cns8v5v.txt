    88	
    89	  const handleFormSubmit = (e: React.FormEvent<HTMLFormElement>) => {
    90	    e.preventDefault();

    92	    // namedItem returns Element | RadioNodeList | null. The cast keeps `| null` rather than
    93	    // asserting it away — these fields are always single <input>s rendered by ZimmTextInput,

   275	type ContactIconRowProps = {

   280	
   281	type ContactLink = { href?: string; label: string; icon: 'phone' | 'whatsapp' | 'email' };
   282	
   283	/**
   284	 * True when following the href would navigate away from the storefront.
   285	 *
   286	 * Takes `string | undefined` rather than `string` so the caller does not need an `as string`
   287	 * to satisfy it inside an isSafeHref branch. isSafeHref returns a plain boolean, which carries
   288	 * no narrowing, and a `href is string` predicate would be unsound — an unsafe value like
   289	 * `javascript:void(0)` is a string that returns false, so the else branch would narrow to
   290	 * `undefined` incorrectly. An absent href never leaves the site, so `?? ''` is the honest answer.
   291	 *
   292	 * Derived from the scheme rather than keyed to a channel name: tel: and mailto: hand off to
   293	 * the OS dialer or mail client and never navigate, so a target on them does nothing. An
   294	 * http(s) channel — WhatsApp today — is a real cross-origin navigation and opens in a new
   295	 * tab so a shopper mid-browse does not lose the site. Any future http(s) channel inherits
   296	 * this without another edit here.
   297	 */
   298	const leavesTheSite = (href: string | undefined): boolean => /^https?:/i.test(href ?? '');
   299	
   300	/** WCAG 3.2.5 — a link that opens a new tab has to say so in its accessible name. */
   301	const NEW_TAB_SUFFIX = ' — opens in a new tab';
   302	
   303	const ContactIconRow = ({ phoneHref, whatsAppHref, emailHref }: ContactIconRowProps) => {
   304	  const styles = useMultiStyleConfig(ZIMM_FOOTER_RECIPE, {});
   305	  const links: ContactLink[] = [
   306	    { href: phoneHref, label: 'Call us', icon: 'phone' },
   307	    { href: whatsAppHref, label: 'WhatsApp', icon: 'whatsapp' },
   308	    { href: emailHref, label: 'Email us', icon: 'email' },
   309	  ];
   310	  return (
   311	    <chakra.div __css={styles.contactRow}>
   312	      {links.map(({ href, label, icon }) =>
   313	        // isSafeHref, not a truthiness check: these arrive from Contentful, so a
   314	        // javascript:/data:/protocol-relative value would otherwise be rendered
   315	        // straight onto an anchor. Rejected values fall through to the span below,
   316	        // which is already the correct treatment for "no usable destination".
   317	        isSafeHref(href) ? (
   318	          <chakra.a
   319	            key={icon}
   320	            href={href}
   321	            // rel is explicit rather than relying on target="_blank" implying noopener:
   322	            // that is browser default behaviour, not a guarantee, and noreferrer also
   323	            // stops the storefront URL leaking to the third party.
   324	            {...(leavesTheSite(href) ? { target: '_blank', rel: 'noopener noreferrer' } : {})}
   325	            aria-label={leavesTheSite(href) ? `${label}${NEW_TAB_SUFFIX}` : label}
   326	            __css={styles.contactIconLink}
   327	          >
   328	            <ZimmIcon type={icon} size="md" aria-hidden={true} />
   329	          </chakra.a>
   330	        ) : (
   331	          // Fallback for a channel with no usable destination — either the field is absent
   332	          // for this storefront, or isSafeHref rejected the value. Both reach here, and both
   333	          // want the same treatment. NON-INTERACTIVE but PERCEIVABLE:
   334	          //   - <span>: no href, no tabIndex, so nothing promises an action that cannot be
   335	          //     performed (avoids the WCAG 2.4.4 / 3.2.4 false affordance).
   336	          //   - labelled, not aria-hidden: ZimmIcon emits role="img" + aria-label, so screen
   337	          //     reader users learn the channel exists instead of getting silence (avoids the
   338	          //     1.1.1 information loss the previous aria-hidden caused).
   339	          // Not dead code: contact channels are optional per storefront, so a locale that
   340	          // supplies only e.g. email still renders the other two icons this way.
   341	          <chakra.span key={icon} __css={styles.contactIconLink}>
   342	            <ZimmIcon type={icon} size="md" aria-label={label} />
   343	          </chakra.span>
   344	        ),
   345	      )}
   346	    </chakra.div>
   347	  );
   348	};

   476	                        key={link.sys.id}
   477	                        __css={{
   478	                          ...styles.linkListItem,
   479	                          ...(group.fields.showContactChannels ? { paddingLeft: '2xs' } : {}),
   480	                        }}
   481	                      >
   482	                        {/* Each item h-[24px] items-center, no gap. ZimmNavItem's baseStyle supplies
   483	                            minHeight 24px, whiteSpace normal (so the label below can wrap) and uppercase.
   484	
   485	                            width:min-content only for labels that carry a non-breaking space. A wrapped

   494	                          href={link.fields.href}

   500	                    ))}
   501	                    {/* Contact channels are the LAST CHILD OF THE LIST, not a sibling of it —
   502	                        Figma's Icon-list (40002081:14616) sits inside Links at y=72, directly

   506	                    {group.fields.showContactChannels && (
   507	                      <chakra.li __css={styles.linkListItem}>
   508	                        <ContactIconRow phoneHref={phoneHref} whatsAppHref={whatsAppHref} emailHref={emailHref} />

   514	            </chakra.div>
   515	          </chakra.div>
   516	        </chakra.div>
   517	
   518	        <chakra.div __css={styles.logo}>
   519	          {/* Button + programmatic nav, matching the header's wordmark rather than an <a href>.

   523	          <chakra.button type="button" aria-label="Zimmermann home" onClick={onLogoClick} __css={styles.logoButton}>
   524	            <ZimmLogo variant="inverse" />
   525	          </chakra.button>
   526	        </chakra.div>
   527	      </chakra.div>
   528	
   529	      <BottomBar copyrightYear={copyrightYear} copyrightOwner={copyrightOwner ?? ''} legalLinks={legalItems} />
   530	    </chakra.footer>
   531	  );
   532	};

   538	/**
   539	 * The component SFCC actually renders. Base `_app/index.jsx` resolves
   540	 * `@salesforce/retail-react-app/app/components/footer` through the overrides system
   541	 * and renders `<Footer />` with **no props**, so there is no upstream container to
   542	 * own the data. This wrapper is that seam, and the only place static content lives.
   543	 *
   544	 * HPC-034: replace FOOTER_FIXTURE with the Contentful CDA fetch **here** — the
   545	 * presentational Footer above does not change. Note that puts a fetch inside
   546	 * `components/`, which components.md reserves for containers; if that matters,
   547	 * HPC-034 should override `_app` instead (Tier 2 shared shell — coordinate first).
   548	 */
   549	const FooterContainer = (props: Omit<FooterProps, 'data'>) => {
   550	  // buildUrl + history.push, not useNavigation. useNavigation special-cases the home path
   551	  // (`path === '/' ? '/' : updatedHref`), so navigate('/') pushes a bare '/' and skips the
   552	  // site/locale prefix — on a prefixed storefront that drops the shopper onto the default
   553	  // site. This is the exact pair base _app's onLogoClick uses, so both wordmarks resolve
   554	  // home the same way on every storefront.
   555	  const { buildUrl } = useMultiSite();
   556	  const history = useHistory();
   557	  return <Footer data={FOOTER_FIXTURE} onLogoClick={() => history.push(buildUrl(HOME_HREF))} {...props} />;
   558	};
