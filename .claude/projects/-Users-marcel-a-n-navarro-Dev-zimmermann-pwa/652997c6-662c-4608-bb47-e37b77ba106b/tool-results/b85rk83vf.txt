    63	type SubscribeModalProps = {
    64	  isOpen: boolean;
    65	  onClose: () => void;
    66	  onSubmit: (firstName: string, lastName: string) => void;

    70	const SubscribeModal = ({ isOpen, onClose, onSubmit, privacyPolicyHref }: SubscribeModalProps) => {
    71	  const [mounted, setMounted] = useState(false);
    72	  const formRef = useRef<HTMLFormElement>(null);
    73	  const firstFocusRef = useRef<HTMLDivElement>(null);
    74	  const triggerRef = useRef<Element | null>(null);
    75	  const styles = useMultiStyleConfig('ZimmFooter', {});
    76	
    77	  useEffect(() => {
    78	    setMounted(true);
    79	  }, []);
    80	
    81	  // Save the trigger element on open so focus can be restored on close
    82	  useEffect(() => {
    83	    if (isOpen) {
    84	      triggerRef.current = document.activeElement;
    85	      // Defer so the portal has rendered before we try to focus inside it
    86	      const id = setTimeout(() => {
    87	        const firstInput = firstFocusRef.current?.querySelector<HTMLElement>('input');
    88	        firstInput?.focus();
    89	      }, 0);
    90	      return () => clearTimeout(id);
    91	    } else {
    92	      // Return focus to the element that opened the modal
    93	      if (triggerRef.current instanceof HTMLElement) {
    94	        triggerRef.current.focus();
    95	      }
    96	    }
    97	  }, [isOpen]);
    98	
    99	  // Lock body scroll while modal is open
   100	  useEffect(() => {
   101	    if (!isOpen) return;
   102	    const prev = document.body.style.overflow;
   103	    document.body.style.overflow = 'hidden';
   104	    return () => {
   105	      document.body.style.overflow = prev;
   106	    };
   107	  }, [isOpen]);
   108	
   109	  // Keyboard: Escape to close, Tab trap within modal
   110	  const handleKeyDown = useCallback(
   111	    (e: KeyboardEvent) => {
   112	      if (e.key === 'Escape') {
   113	        onClose();
   114	        return;
   115	      }
   116	      if (e.key !== 'Tab' || !formRef.current) return;
   117	
   118	      const focusable = Array.from(
   119	        formRef.current.querySelectorAll<HTMLElement>(
   120	          'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])',
   121	        ),
   122	      ).filter((el) => !el.hasAttribute('disabled'));
   123	
   124	      const first = focusable[0];
   125	      const last = focusable[focusable.length - 1];
   126	
   127	      if (e.shiftKey && document.activeElement === first) {
   128	        e.preventDefault();
   129	        last?.focus();
   130	      } else if (!e.shiftKey && document.activeElement === last) {
   131	        e.preventDefault();
   132	        first?.focus();
   133	      }
   134	    },
   135	    [onClose],
   136	  );
   137	
   138	  useEffect(() => {
   139	    if (!isOpen) return;
   140	    document.addEventListener('keydown', handleKeyDown);
   141	    return () => document.removeEventListener('keydown', handleKeyDown);
   142	  }, [isOpen, handleKeyDown]);
   143	
   144	  const handleFormSubmit = (e: React.FormEvent<HTMLFormElement>) => {
   145	    e.preventDefault();
   146	    const form = e.currentTarget;
   147	    const firstName = (form.elements.namedItem('firstName') as HTMLInputElement)?.value ?? '';
   148	    const lastName = (form.elements.namedItem('lastName') as HTMLInputElement)?.value ?? '';
   149	    onSubmit(firstName.trim(), lastName.trim());
   150	  };
   151	
   152	  const handleBackdropClick = (e: React.MouseEvent<HTMLDivElement>) => {
   153	    if (e.target === e.currentTarget) onClose();
   154	  };
   155	
   156	  if (!mounted || !isOpen) return null;
   157	
   158	  return createPortal(
   159	    <chakra.div role="presentation" onClick={handleBackdropClick} __css={styles.modalOverlay}>
   160	      <chakra.div
   161	        role="dialog"
   162	        aria-modal="true"
   163	        aria-labelledby="zimm-subscribe-modal-title"
   164	        aria-describedby="zimm-subscribe-modal-consent"
   165	        __css={styles.subscribeModalPanel}
   166	      >
   167	        <form ref={formRef} onSubmit={handleFormSubmit} noValidate>
   168	          {/* Title */}
   169	          <chakra.h2 id="zimm-subscribe-modal-title" __css={styles.modalTitle}>
   170	            Join Us
   171	          </chakra.h2>
   172	
   173	          {/* First Name */}
   174	          <chakra.div ref={firstFocusRef} __css={styles.modalField}>
   175	            <ZimmTextInput label="First Name" name="firstName" />
   176	          </chakra.div>
   177	
   178	          {/* Last Name */}
   179	          <chakra.div __css={styles.modalFieldLast}>
   180	            <ZimmTextInput label="Last Name" name="lastName" />
   181	          </chakra.div>
   182	
   183	          {/* Privacy consent copy (HPC-037) */}
   184	          <chakra.p id="zimm-subscribe-modal-consent" __css={styles.modalConsent}>
   185	            {PRIVACY_CONSENT_COPY}{' '}
   186	            {/* Same guard as the contact channels: privacyPolicyHref is a prop, so a caller
   187	                can supply anything. Unsafe or absent renders the label as plain text rather
   188	                than an anchor that goes nowhere. */}
   189	            {isSafeHref(privacyPolicyHref) ? (
   190	              <chakra.a href={privacyPolicyHref} __css={styles.modalConsentLink}>
   191	                Privacy Policy
   192	              </chakra.a>
   193	            ) : (
   194	              <chakra.span __css={styles.modalConsentLink}>Privacy Policy</chakra.span>
   195	            )}
   196	            .
   197	          </chakra.p>
   198	
   199	          {/* Accessibility disclaimer (HPC-037) */}
   200	          <chakra.p __css={styles.modalDisclaimer}>{ACCESSIBILITY_DISCLAIMER_COPY}</chakra.p>
   201	
   202	          {/* Actions */}
   203	          <chakra.div __css={styles.modalActions}>
   204	            <ZimmButton variant="secondary" size="medium" label="Cancel" type="button" onClick={onClose} />
   205	            <ZimmButton variant="primary" size="medium" label="Continue" type="submit" />
   206	          </chakra.div>
   207	        </form>
   208	      </chakra.div>
   209	    </chakra.div>,
   210	    document.body,
   211	  );
   212	};

   221	// The HPC-036 sign-up payload: footer email (HPC-034) + modal names (HPC-035).

   225	type SubscribeSectionProps = { heading: string; body: string; initialSubscribed?: boolean; privacyPolicyHref: string };
   226	
   227	const SUBSCRIBE_SUCCESS_MESSAGE = "You're on the list! Be the first to know about new collections and new arrivals.";
   228	const SUBSCRIBE_UNAVAILABLE_MESSAGE = 'Subscription is temporarily unavailable. Please try again later.';
   229	
   230	const SubscribeSection = ({ heading, body, initialSubscribed = false, privacyPolicyHref }: SubscribeSectionProps) => {
   231	  // Map the Storybook-only `initialSubscribed` flag to the success state.
   232	  // Post HPC-036, 'success' will only be reached via a confirmed Omneo response.
   233	  const [submitState, setSubmitState] = useState<SubmitState>(initialSubscribed ? 'success' : 'idle');
   234	  // Ref, not state: nothing renders from it, so a re-render on capture would be wasted.
   235	  // The footer input unmounts once submitState leaves 'idle', so the email must be
   236	  // captured here at step 1 or it is unrecoverable at step 2.
   237	  const pendingSignupRef = useRef<PendingSignup>({ email: '', firstName: '', lastName: '' });
   238	  const [modalOpen, setModalOpen] = useState(false);
   239	  const styles = useMultiStyleConfig('ZimmFooter', {});
   240	
   241	  const handleEmailSubmit = (email: string) => {
   242	    pendingSignupRef.current.email = email;
   243	    setModalOpen(true);
   244	  };
   245	
   246	  const handleModalSubmit = (firstName: string, lastName: string) => {
   247	    pendingSignupRef.current = { ...pendingSignupRef.current, firstName, lastName };
   248	    // PLACEHOLDER until HPC-036 (Omneo integration) lands: POST pendingSignupRef.current here.
   249	    // On confirmed response → setSubmitState('success'). On failure → setSubmitState('error').
   250	    // 'unavailable' is shown until the integration is live; never show false success.
   251	    setModalOpen(false);
   252	    setSubmitState('unavailable');
   253	    // Nothing was sent, so nothing needs retaining — the record holds customer PII and there
   254	    // is no request for it to outlive. When HPC-036 adds the POST above, this reset stays; it
   255	    // just moves to after a confirmed response. React memory only, never storage, never logged.
   256	    pendingSignupRef.current = { email: '', firstName: '', lastName: '' };
   257	  };
   258	
   259	  return (
   260	    <div>
   261	      <chakra.p __css={styles.subscribeHeading}>{heading}</chakra.p>
   262	      <chakra.p __css={styles.subscribeBody}>{body}</chakra.p>
   263	      <div aria-live="polite">
   264	        {submitState === 'idle' && (
   265	          <ZimmInputAction
   266	            label="Email"
   267	            type="email"
   268	            name="footer-email"
   269	            icon="chevron-right"
   270	            buttonLabel="Subscribe"
   271	            inverse
   272	            onSubmit={handleEmailSubmit}
   273	          />
   274	        )}
   275	        {submitState === 'success' && <chakra.p __css={styles.subscribeSuccess}>{SUBSCRIBE_SUCCESS_MESSAGE}</chakra.p>}
   276	        {(submitState === 'unavailable' || submitState === 'error') && (
   277	          <chakra.p __css={styles.subscribeSuccess}>{SUBSCRIBE_UNAVAILABLE_MESSAGE}</chakra.p>
   278	        )}
   279	      </div>
   280	
   281	      <SubscribeModal
   282	        isOpen={modalOpen}
   283	        onClose={() => setModalOpen(false)}
   284	        onSubmit={handleModalSubmit}
   285	        privacyPolicyHref={privacyPolicyHref}
   286	      />
   287	    </div>
   288	  );
   289	};

   292	// Contact icon row — rendered below the link list in the Contact column
   293	// ---------------------------------------------------------------------------
   294	// Figma: Icon-list has pt-4px (alias/space/100) above it and 4px internal gap between icons.
   295	// hrefs sourced from Contentful FooterConfigEntry (HPC-034). Until that integration lands,
   296	// all hrefs are undefined and each icon renders as a non-interactive but labelled <span> —
   297	// visible in the layout and announced by screen readers, but absent from the tab order.
   298	// See the inline note on the fallback branch below for the full a11y reasoning.
   299	// When a href arrives it switches to a live <a>, no other changes needed.
   300	type ContactIconRowProps = {
   301	  phoneHref?: string; // locale-aware tel: href from Contentful
   302	  whatsAppHref?: string; // WhatsApp URL from Contentful
   303	  emailHref?: string; // mailto: address from Contentful
   304	};
   305	
   306	type ContactLink = { href?: string; label: string; icon: 'phone' | 'whatsapp' | 'email' };
   307	
   308	/**
   309	 * True when following the href would navigate away from the storefront.
   310	 *
   311	 * Derived from the scheme rather than keyed to a channel name: tel: and mailto: hand off to
   312	 * the OS dialer or mail client and never navigate, so a target on them does nothing. An
   313	 * http(s) channel — WhatsApp today — is a real cross-origin navigation and opens in a new
   314	 * tab so a shopper mid-browse does not lose the site. Any future http(s) channel inherits
   315	 * this without another edit here.
   316	 */
   317	const leavesTheSite = (href: string): boolean => /^https?:/i.test(href);
   318	
   319	/** WCAG 3.2.5 — a link that opens a new tab has to say so in its accessible name. */
   320	const NEW_TAB_SUFFIX = ' — opens in a new tab';
   321	
   322	const ContactIconRow = ({ phoneHref, whatsAppHref, emailHref }: ContactIconRowProps) => {
   323	  const styles = useMultiStyleConfig('ZimmFooter', {});
   324	  const links: ContactLink[] = [
   325	    { href: phoneHref, label: 'Call us', icon: 'phone' },
   326	    { href: whatsAppHref, label: 'WhatsApp', icon: 'whatsapp' },
   327	    { href: emailHref, label: 'Email us', icon: 'email' },
   328	  ];
   329	  return (
   330	    <chakra.div __css={styles.contactRow}>
   331	      {links.map(({ href, label, icon }) =>
   332	        // isSafeHref, not a truthiness check: these arrive from Contentful, so a
   333	        // javascript:/data:/protocol-relative value would otherwise be rendered
   334	        // straight onto an anchor. Rejected values fall through to the span below,
   335	        // which is already the correct treatment for "no usable destination".
   336	        isSafeHref(href) ? (
   337	          <chakra.a
   338	            key={icon}
   339	            href={href}
   340	            // rel is explicit rather than relying on target="_blank" implying noopener:
   341	            // that is browser default behaviour, not a guarantee, and noreferrer also
   342	            // stops the storefront URL leaking to the third party.
   343	            {...(leavesTheSite(href as string) ? { target: '_blank', rel: 'noopener noreferrer' } : {})}
   344	            aria-label={leavesTheSite(href as string) ? `${label}${NEW_TAB_SUFFIX}` : label}
   345	            __css={styles.contactIconLink}
   346	          >
   347	            <ZimmIcon type={icon} size="md" aria-hidden={true} />
   348	          </chakra.a>
   349	        ) : (
   350	          // Fallback for a channel with no usable destination — either the field is absent
   351	          // for this storefront, or isSafeHref rejected the value. Both reach here, and both
   352	          // want the same treatment. NON-INTERACTIVE but PERCEIVABLE:
   353	          //   - <span>: no href, no tabIndex, so nothing promises an action that cannot be
   354	          //     performed (avoids the WCAG 2.4.4 / 3.2.4 false affordance).
   355	          //   - labelled, not aria-hidden: ZimmIcon emits role="img" + aria-label, so screen
   356	          //     reader users learn the channel exists instead of getting silence (avoids the
   357	          //     1.1.1 information loss the previous aria-hidden caused).
   358	          // Not dead code: contact channels are optional per storefront, so a locale that
   359	          // supplies only e.g. email still renders the other two icons this way.
   360	          <chakra.span key={icon} __css={styles.contactIconLink}>
   361	            <ZimmIcon type={icon} size="md" aria-label={label} />
   362	          </chakra.span>
   363	        ),
   364	      )}
   365	    </chakra.div>
   366	  );
   367	};

   555	/**
   556	 * The component SFCC actually renders. Base `_app/index.jsx` resolves
   557	 * `@salesforce/retail-react-app/app/components/footer` through the overrides system
   558	 * and renders `<Footer />` with **no props**, so there is no upstream container to
   559	 * own the data. This wrapper is that seam, and the only place static content lives.
   560	 *
   561	 * HPC-034: replace FOOTER_FIXTURE with the Contentful CDA fetch **here** — the
   562	 * presentational Footer above does not change. Note that puts a fetch inside
   563	 * `components/`, which components.md reserves for containers; if that matters,
   564	 * HPC-034 should override `_app` instead (Tier 2 shared shell — coordinate first).
   565	 */
   566	const FooterContainer = (props: Omit<FooterProps, 'data'>) => {
   567	  // buildUrl + history.push, not useNavigation. useNavigation special-cases the home path
   568	  // (`path === '/' ? '/' : updatedHref`), so navigate('/') pushes a bare '/' and skips the
   569	  // site/locale prefix — on a prefixed storefront that drops the shopper onto the default
   570	  // site. This is the exact pair base _app's onLogoClick uses, so both wordmarks resolve
   571	  // home the same way on every storefront.
   572	  const { buildUrl } = useMultiSite();
   573	  const history = useHistory();
   574	  return <Footer data={FOOTER_FIXTURE} onLogoClick={() => history.push(buildUrl(HOME_HREF))} {...props} />;
   575	};
