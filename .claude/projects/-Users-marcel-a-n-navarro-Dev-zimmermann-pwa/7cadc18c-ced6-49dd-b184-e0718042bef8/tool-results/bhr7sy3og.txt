   207	 * variant (see ZimmDrawerSize), built on Chakra's Drawer primitive (focus trap, initial focus,
   208	 * focus restore, body-scroll lock, and Escape/overlay-click-to-close all come from Chakra, not
   209	 * hand-rolled).
   210	 *
   211	 * Pass `title` to render an inline header row (title text left, close button right) sized per the
   212	 * Figma drawer/header frames. Omit `title` and pass `labelledBy`/`aria-label` instead for drawers
   213	 * that supply their own header (e.g. ZimmPdpDrawer's tab row) — the close button floats on its own
   214	 * by default. Add `boxedHeader` alongside `labelledBy` when the drawer has no header content of its
   215	 * own (e.g. ZimmCountryLanguageMismatch, whose heading lives in its body) but still wants the close
   216	 * button sitting in that same in-flow header row rather than floating over the content.
   217	 *
   218	 * @param props see ZimmDrawerProps
   219	 */
   220	export const ZimmDrawer = (props: ZimmDrawerProps) => {
   221	  const { isOpen, onClose, size, children, titleStyle = 'large', finalFocusRef, initialFocusRef } = props;
   222	  const styles = useMultiStyleConfig(ZIMM_DRAWER_RECIPE, { variant: size });
   223	  const titleId = useId();
   224	
   225	  const title = 'title' in props ? props.title : undefined;
   226	  const labelledBy = !title && 'labelledBy' in props ? props.labelledBy : undefined;
   227	  const ariaLabel = !title && 'aria-label' in props ? props['aria-label'] : undefined;
   228	  const boxedHeader = !title && 'labelledBy' in props ? props.boxedHeader : undefined;
   229	
   230	  // SFCC-926 — tester reported body scroll still active while any drawer is open, allowing
   231	  // the page behind the overlay to scroll alongside the drawer. ZimmDrawer bypasses Chakra's
   232	  // own <DrawerContent>, so blockScrollOnMount may not fire. Locking body overflow explicitly
   233	  // ensures only the drawer's internal scroll area is accessible while it is open.
   234	  // Uses the module-level counter so multiple open instances don't race to restore overflow.
   235	  useEffect(() => {
   236	    if (!isOpen) return;
   237	    if (scrollLockCount === 0) {
   238	      scrollLockPrev = document.body.style.overflow;
   239	      document.body.style.overflow = 'hidden';
   240	    }
   241	    scrollLockCount++;
   242	    return () => {
   243	      scrollLockCount--;
   244	      if (scrollLockCount === 0) {
   245	        document.body.style.overflow = scrollLockPrev;
   246	        scrollLockPrev = '';
   247	      }
   248	    };
   249	  }, [isOpen]);
   250	
   251	  // Chakra's FocusLock remains active during the exit animation, so any focus call made
   252	  // while the drawer is still mounted (rAF, close-button onClick) gets trapped and dropped.
   253	  // onCloseComplete fires after the animation finishes and FocusLock is released — the only
   254	  // reliable moment to restore focus. Covers both close-button click and Escape for all
   255	  // consumers that pass finalFocusRef.
   256	  const handleCloseComplete = () => {
   257	    if (finalFocusRef) finalFocusRef.current?.focus();
   258	  };
   259	
   260	  return (
   261	    <Drawer
   262	      isOpen={isOpen}
   263	      onClose={onClose}
   264	      placement="right"
   265	      onCloseComplete={handleCloseComplete}
   266	      initialFocusRef={initialFocusRef}
   267	    >
   268	      <DrawerOverlay sx={styles.overlay} />
   269	      {/* aria-labelledby (not aria-label) for the title case: Chakra's own <DrawerContent>
   270	          would clobber it with its own headingId (undefined without a DrawerHeader), but
   271	          ZimmDrawerContent bypasses <DrawerContent> entirely and sets dialogProps itself
   272	          (see its own comment above) — the clobbering bug this would otherwise work around
   273	          doesn't apply here, so referencing the visible title node is safe and preferred. */}
   274	      <ZimmDrawerContent
   275	        onClose={onClose}
   276	        title={title}
   277	        titleId={title ? titleId : undefined}
   278	        titleStyle={titleStyle}
   279	        labelledBy={labelledBy}
   280	        ariaLabel={ariaLabel}
   281	        boxedHeader={boxedHeader}
   282	        contentStyles={styles.content}
   283	        headerStyles={styles.header}
   284	        headerTitleStyles={styles.headerTitle}
   285	        closeButtonStyles={styles.closeButton}
   286	        headerCloseButtonStyles={styles.headerCloseButton}
   287	      >
   288	        {children}
   289	      </ZimmDrawerContent>
   290	    </Drawer>
   291	  );
   292	};

   132	 * Renders the inline header row (title left — empty when there's no `title` text — close button
   133	 * right) when either `title` or `boxedHeader` is passed, or a floating close button only when
   134	 * neither is — see ZimmDrawer's own docblock for when each is used.
   135	 *
   136	 * @param props see ZimmDrawerContentConfig
   137	 */
   138	const ZimmDrawerContent = ({
   139	  children,
   140	  onClose,
   141	  title,
   142	  titleId,
   143	  titleStyle = 'large',
   144	  labelledBy,
   145	  ariaLabel,
   146	  boxedHeader,
   147	  contentStyles,
   148	  headerStyles,
   149	  headerTitleStyles,
   150	  closeButtonStyles,
   151	  headerCloseButtonStyles,
   152	}: ZimmDrawerContentConfig) => {
   153	  const { t } = useUiCopy();
   154	  const { getDialogProps, getDialogContainerProps } = useModalContext();
   155	  const dialogProps = getDialogProps({ 'aria-label': ariaLabel });
   156	  // getDialogProps always computes `aria-labelledby` from Chakra's own <DrawerHeader> mount
   157	  // state, which ZimmDrawer never renders (no header sub-component in this tree) — so it
   158	  // always resolves to undefined regardless of what's passed in above. Set it directly so the
   159	  // `title` and `labelledBy` prop variants of ZimmDrawerProps actually work.
   160	  if (title && titleId) {
   161	    dialogProps['aria-labelledby'] = titleId;
   162	  } else if (labelledBy) {
   163	    dialogProps['aria-labelledby'] = labelledBy;
   164	  }
   165	  const containerProps = getDialogContainerProps();
   166	
   167	  return (
   168	    <ModalFocusScope>
   169	      <chakra.div {...containerProps} __css={{ position: 'fixed', inset: 0, display: 'flex', zIndex: 'modal' }}>
   170	        <MotionChakraDiv
   171	          {...dialogProps}
   172	          initial={{ x: '100%' }}
   173	          animate={{ x: 0 }}
   174	          exit={{ x: '100%' }}
   175	          transition={DRAWER_TRANSITION}
   176	          __css={{ position: 'fixed', top: 0, right: 0, bottom: 0, outline: 0, ...contentStyles }}
   177	        >
   178	          {title || boxedHeader ? (
   179	            <chakra.div sx={headerStyles}>
   180	              <chakra.span
   181	                id={title ? titleId : undefined}
   182	                sx={headerTitleStyles}
   183	                textStyle={titleStyle === 'small' ? 'body2Strong' : 'heading1'}
   184	              >
   185	                {title}
   186	              </chakra.span>
   187	              <chakra.div sx={headerCloseButtonStyles}>
   188	                <ZimmButton variant="icon" icon="close" label={t('drawer.close', 'Close')} onClick={onClose} />
   189	              </chakra.div>
   190	            </chakra.div>
   191	          ) : (
   192	            <chakra.div sx={closeButtonStyles}>
   193	              <ZimmButton variant="icon" icon="close" label={t('drawer.close', 'Close')} onClick={onClose} />
   194	            </chakra.div>
   195	          )}
   196	          {/* SFCC-926 — tester reported no scrollbar when drawer content exceeds the panel
   197	              height. flex:1 + minHeight:0 lets this grow into remaining space after the
   198	              fixed header; overflowY:auto gives it its own independent scrollbar. */}
   199	          <chakra.div __css={{ flex: 1, overflowY: 'auto', minHeight: 0 }}>
   200	            {children}
   201	          </chakra.div>
   202	        </MotionChakraDiv>
   203	      </chakra.div>
   204	    </ModalFocusScope>
   205	  );
   206	};
   207	
   208	/**
   209	 * Generic right-anchored slide-in drawer shell — overlay, panel, and a Figma-matched size
   210	 * variant (see ZimmDrawerSize), built on Chakra's Drawer primitive (focus trap, initial focus,
   211	 * focus restore, body-scroll lock, and Escape/overlay-click-to-close all come from Chakra, not
   212	 * hand-rolled).
   213	 *
   214	 * Pass `title` to render an inline header row (title text left, close button right) sized per the
   215	 * Figma drawer/header frames. Omit `title` and pass `labelledBy`/`aria-label` instead for drawers
   216	 * that supply their own header (e.g. ZimmPdpDrawer's tab row) — the close button floats on its own
   217	 * by default. Add `boxedHeader` alongside `labelledBy` when the drawer has no header content of its
   218	 * own (e.g. ZimmCountryLanguageMismatch, whose heading lives in its body) but still wants the close
   219	 * button sitting in that same in-flow header row rather than floating over the content.
   220	 *
   221	 * @param props see ZimmDrawerProps
   222	 */
   223	export const ZimmDrawer = (props: ZimmDrawerProps) => {
   224	  const {
   225	    isOpen,
   226	    onClose,
   227	    size,
   228	    children,
   229	    titleStyle = 'large',
   230	    finalFocusRef,
   231	    initialFocusRef,
   232	    hasOverlay = true,
   233	  } = props;
   234	  const styles = useMultiStyleConfig(ZIMM_DRAWER_RECIPE, { variant: size });
   235	  const titleId = useId();
   236	
   237	  const title = 'title' in props ? props.title : undefined;
   238	  const labelledBy = !title && 'labelledBy' in props ? props.labelledBy : undefined;
   239	  const ariaLabel = !title && 'aria-label' in props ? props['aria-label'] : undefined;
   240	  const boxedHeader = !title && 'labelledBy' in props ? props.boxedHeader : undefined;
   241	
   242	  // SFCC-926 — tester reported body scroll still active while any drawer is open, allowing
   243	  // the page behind the overlay to scroll alongside the drawer. ZimmDrawer bypasses Chakra's
   244	  // own <DrawerContent>, so blockScrollOnMount may not fire. Locking body overflow explicitly
   245	  // ensures only the drawer's internal scroll area is accessible while it is open.
   246	  // Uses the module-level counter so multiple open instances don't race to restore overflow.
   247	  useEffect(() => {
   248	    if (!isOpen) return;
   249	    if (scrollLockCount === 0) {
   250	      scrollLockPrev = document.body.style.overflow;
   251	      document.body.style.overflow = 'hidden';
   252	    }
   253	    scrollLockCount++;
   254	    return () => {
   255	      scrollLockCount--;
   256	      if (scrollLockCount === 0) {
   257	        document.body.style.overflow = scrollLockPrev;
   258	        scrollLockPrev = '';
   259	      }
   260	    };
   261	  }, [isOpen]);
   262	
   263	  // Chakra's FocusLock remains active during the exit animation, so any focus call made
   264	  // while the drawer is still mounted (rAF, close-button onClick) gets trapped and dropped.
   265	  // onCloseComplete fires after the animation finishes and FocusLock is released — the only
   266	  // reliable moment to restore focus. Covers both close-button click and Escape for all
   267	  // consumers that pass finalFocusRef.
   268	  const handleCloseComplete = () => {
   269	    if (finalFocusRef) finalFocusRef.current?.focus();
   270	  };
   271	
   272	  return (
   273	    <Drawer
   274	      isOpen={isOpen}
   275	      onClose={onClose}
   276	      placement="right"
   277	      onCloseComplete={handleCloseComplete}
   278	      initialFocusRef={initialFocusRef}
   279	    >
   280	      {hasOverlay && <DrawerOverlay sx={styles.overlay} />}
   281	      {/* aria-labelledby (not aria-label) for the title case: Chakra's own <DrawerContent>
   282	          would clobber it with its own headingId (undefined without a DrawerHeader), but
   283	          ZimmDrawerContent bypasses <DrawerContent> entirely and sets dialogProps itself
   284	          (see its own comment above) — the clobbering bug this would otherwise work around
   285	          doesn't apply here, so referencing the visible title node is safe and preferred. */}
   286	      <ZimmDrawerContent
   287	        onClose={onClose}
   288	        title={title}
   289	        titleId={title ? titleId : undefined}
   290	        titleStyle={titleStyle}
   291	        labelledBy={labelledBy}
   292	        ariaLabel={ariaLabel}
   293	        boxedHeader={boxedHeader}
   294	        contentStyles={styles.content}
   295	        headerStyles={styles.header}
   296	        headerTitleStyles={styles.headerTitle}
   297	        closeButtonStyles={styles.closeButton}
   298	        headerCloseButtonStyles={styles.headerCloseButton}
   299	      >
   300	        {children}
   301	      </ZimmDrawerContent>
   302	    </Drawer>
   303	  );
   304	};
