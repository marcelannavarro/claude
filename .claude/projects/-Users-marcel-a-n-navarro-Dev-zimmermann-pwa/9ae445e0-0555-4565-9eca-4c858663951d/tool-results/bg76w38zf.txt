+import React, { useState } from 'react';
+import { chakra, useMultiStyleConfig } from '@chakra-ui/react';
+import { ZimmTextInput } from 'app/components/ui-kits/atoms/zimmTextInput';
+import { ZimmButton } from 'app/components/ui-kits/atoms/zimmButton';
+import { useUiCopy } from 'app/hooks/useUiCopy';
+
+const ZIMM_AUTH_DRAWER_CONTENT_RECIPE = 'ZimmAuthDrawerContent';
+
+const VALIDATION_EMPTY = 'Please enter a value.';
+const VALIDATION_EMAIL = 'Please enter a valid email address.';
+// Matches HTML5 type="email" validity behaviour for the common case.
+const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
+
+type ApiResult = { success: boolean; error?: string };
+
+type EmailFormProps = {
+  emailValue: string;
+  onEmailChange: (val: string) => void;
+  /** May return an ApiResult — if success is false and error is set, the error is shown in the input. */
+  onSubmit: () => Promise<ApiResult | void> | void;
+  isLoading?: boolean;
+  /** Forwarded to the email input for programmatic focus (e.g. drawer initialFocusRef). */
+  inputRef?: React.RefObject<HTMLInputElement>;
+};
+
+export const EmailForm = ({ emailValue, onEmailChange, onSubmit, isLoading = false, inputRef }: EmailFormProps) => {
+  const { t } = useUiCopy();
+  const emailLabel = t('login.drawer.email', 'Email');
+  const sendCode = t('login.drawer.sendCode', 'Send code');
+  const styles = useMultiStyleConfig(ZIMM_AUTH_DRAWER_CONTENT_RECIPE, {});
+  const [validationError, setValidationError] = useState<string | undefined>();
+
+  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
+    e.preventDefault();
+    const trimmed = emailValue.trim();
+    if (!trimmed) {
+      setValidationError(VALIDATION_EMPTY);
+      return;
+    }
+    if (!EMAIL_RE.test(trimmed)) {
+      setValidationError(VALIDATION_EMAIL);
+      return;
+    }
+    setValidationError(undefined);
+    const result = await onSubmit();
+    if (result && !result.success && result.error) {
+      setValidationError(result.error);
+    }
+  };
+
+  return (
+    <chakra.form __css={styles.emailFormStack} onSubmit={handleSubmit} noValidate>
+      <ZimmTextInput
+        ref={inputRef}
+        label={emailLabel}
+        type="email"
+        name="email"
+        value={emailValue}
+        onChange={(val) => {
+          setValidationError(undefined);
+          onEmailChange(val);
+        }}
+        errorText={validationError}
+        outlined
+      />
+      <ZimmButton
+        variant="primary"
+        size="adaptive"
+        label={sendCode}
+        fullWidth
+        type="submit"
+        disabled={isLoading}
+      />
+    </chakra.form>
+  );
+};

+    super(props);
+    this.state = { hasError: false };
+  }
+

+    return { hasError: true };
+  }

+  let skip = 0;
+  let total = Infinity;
+  while (existing.length < total) {
+    const page = await client.entry.getMany({
+      environmentId: TARGET,
+      query: { content_type: 'uiCopy', limit: 1000, skip },
+    });
+    total = page.total;
+    existing.push(...page.items);
+    skip += page.items.length;
+    if (page.items.length === 0) break; // guard against a zero-item page to prevent infinite loop
+  }
+
+  // Map from uiCopy key → full entry object so we can inspect sys.publishedVersion below.
+  const existingEntryMap = new Map(
+    existing
+      .map((e) => [e.fields?.key?.['en'], e])
+      .filter(([key]) => Boolean(key)),
+  );
+
+  let created = 0;
+  let skipped = 0;
+  let published = 0;
+
+  for (const entry of ENTRIES) {
+    const existingEntry = existingEntryMap.get(entry.key);
+    if (existingEntry) {
+      if (existingEntry.sys.publishedVersion == null) {
+        // Entry exists as a draft — publish it now so it is available to the app.
+        await client.entry.publish(
+          { environmentId: TARGET, entryId: existingEntry.sys.id },
+          { sys: existingEntry.sys },
+        );
+        log.success(`${entry.key} — draft published`);
+        published++;
+      } else {
+        log.skip(`${entry.key} — already published, skipping`);
+        skipped++;
+      }
+      continue;
+    }
+
+    const created_entry = await client.entry.create(
+      { environmentId: TARGET, contentTypeId: 'uiCopy' },
+      {
+        fields: {
+          title: { 'en': entry.title },
+          key: { 'en': entry.key },
+          value: { 'en': entry.value },
+          description: { 'en': entry.description },
+        },
+      },
+    );
+
+    await client.entry.publish(
+      { environmentId: TARGET, entryId: created_entry.sys.id },
+      { sys: created_entry.sys },
+    );
+
+    log.success(`${entry.key}`);
+    created++;
+  }
+
+  log.section('Done');
+  log.info(`Created and published: ${created}`);
+  if (published) log.info(`Draft entries published: ${published}`);
+  if (skipped) log.info(`Skipped (already published): ${skipped}`);
+}
+
+main().catch((err) => {
+  log.error(err.message ?? String(err));
+  process.exit(1);
+});
