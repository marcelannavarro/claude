/**
 * Exported alongside `ZimmProductCardProps` (not just kept module-private) so a test/fixture can
 * build a properly *narrowed* per-variation props object — e.g.
 * `ZimmProductCardBaseProps & ZimmProductCardWishlistVariation` — instead of a value typed as the
 * full union. This matters for real, not just for convenience: TypeScript's excess-property
 * check on a *union* target is weak (a property valid on any one member is not flagged as excess
 * on a fresh literal discriminated toward a different member), so proving "the wishlist-only
 * fields don't exist on Home" needs an assignment to Home's own narrow shape specifically, not to
 * the union alias — see index.test.tsx's `@ts-expect-error` cases and its overloaded
 * `buildCardProps` helper.
 */
export type ZimmProductCardBaseProps = {
  /** Stable per-card identifier. Not used for any layout/behavioural decision — exposed as a
   *  `data-product-id` attribute on the root element as a DOM hook for QA automation/analytics,
   *  the same kind of caller-facing marker `data-testid` already serves in this build's other
   *  components, without overloading that attribute's own testing-only convention. */
  id: string;
  /** PDP destination, pre-site-prefixed by the caller — forwarded straight to
   *  `ZimmProductCardContent`'s own `href` (see that component's own JSDoc on trust level). */
  href: string;
  name: string;
  headingLevel: ZimmProductCardHeadingLevel;
  primaryImage: ZimmProductCardImage;
  /** Forwarded straight to `ZimmProductCardMedia`'s own same-named prop — see that component's
   *  JSDoc for the full cascading-onError-fallback contract this exists for. */
  primaryImageFallbacks?: ZimmProductCardImage[];
  hoverImage?: ZimmProductCardImage;
  images?: ZimmProductCardImage[];
  currentPrice?: string;
  originalPrice?: string;
  isPriceLoading?: boolean;
  isUnavailable?: boolean;
  /** Only meaningful when `isUnavailable` is true. See the module docblock for why this is not
   *  restricted to the `plp` variation at the type level. */
  onNotifyMe?: () => void;
  /**
   * Merchandiser-configured badge label (e.g. "Runway", "Exclusive"). Always a single value —
   * confirmed via a Figma Dev Mode Connect Note that only the single highest-priority badge is
   * ever shown, never a list (see the module docblock's "Badge count" note). Not restricted to
   * any variation at the type level, same reasoning as `onNotifyMe` above — whether it actually
   * renders is `VARIATION_CONFIG.showBadge`'s job (PLP/Wishlist show it, Home suppresses it; see
   * the module docblock's "Badge" note).
   */
  badge?: string;
  /**
   * SFCC image-sizing/LCP follow-up — forwarded straight to `ZimmProductCardMedia`'s own
   * same-named `priority` prop (its LCP eager-load escape hatch — see that prop's own JSDoc).
   * Not restricted to any variation at the type level, same reasoning as `badge` above: whether
   * a card sits above the fold is a page-composition concern, not a variation concern. Omitted
   * (or `false`) lazy-loads the primary image, this component's unchanged default.
   */
  priority?: boolean;
  /**
   * SFCC image-sizing follow-up — forwarded straight to `ZimmProductCardMedia`'s own same-named
   * `widths` prop. Not restricted to any variation at the type level, same reasoning as `badge`
   * above: PLP, Wishlist, Home, and the Recently Viewed rail each reuse this organism at a
   * different real grid/rail width, so whichever page composes it supplies its own. Omitted
   * renders the raw, unparameterized source exactly as before this prop existed.
   */
  imageWidths?: ZimmMediaTileWidths;
  /** Forwarded straight to `ZimmProductCardMedia`'s own same-named `quality` prop — see that
   *  prop's own JSDoc for the `null`-omits-`&q=` contract. Only used when `imageWidths` is also
   *  given. */
  imageQuality?: number | null;
};

    // Tablet 40000730:13363, Mobile 40000538:25030 — Card-content `hidden="true"` on all three.
    hideBagOnMobile: false,

const messages = defineMessages({
  // Interpolated with the product name, matching ZimmProductPrice's WAS/NOW pattern — Media's
  // own `removeLabel` prop is a plain string, so the interpolation happens here, at the one call
  // site that actually knows the product name.
  removeFromWishlist: {
    id: 'product_card.remove.from_wishlist',
    defaultMessage: 'Remove {name} from wishlist',
  },
});

const ZIMM_PRODUCT_CARD_RECIPE = 'ZimmProductCard';

/**
 * Organism — the full product card, composing `ZimmProductCardMedia` and `ZimmProductCardContent`
 * across the `home` / `plp` / `wishlist` variations. See the module docblock above for the
 * confirmed Figma variation matrix and this component's own recipe (`zimmProductCard.css.ts`) for
 * the stretched-link and hover-reveal mechanics that live entirely in CSS, not in this render
 * body — both children are always rendered in full, with every one of their own sub-parts; CSS
 * alone decides what is visible for a given variation and pointer capability.
 *
 * @param props see `ZimmProductCardProps` — a discriminated union on `variation`.
 */
export const ZimmProductCard = (props: ZimmProductCardProps) => {
  const {
    id,
    href,
    name,
    headingLevel,
    primaryImage,
    primaryImageFallbacks,
    hoverImage,
    images,
    currentPrice,
    originalPrice,
    isPriceLoading,
    isUnavailable,
    onNotifyMe,
    badge,
    variation,
    priority,
    imageWidths,
    imageQuality,
  } = props;

  const { onAddToBag, sizeChipValues, selectedSizeChipValue, onSizeChipSelect, colour, size, onRemove } =
    props as ZimmProductCardVariationFields;

  const intl = useIntl();
  const history = useHistory();
  const config = VARIATION_CONFIG[variation];

  // Was `Boolean(isUnavailable) && Boolean(onNotifyMe)` — changed 2026-09-08, Marcel's explicit
  // decision: show the button whenever a size/product is unavailable even with no real handler
  // wired yet. The real Notify Me feature (PRD-018–021: a slide-out form submitting to a
  // back-in-stock capture target) doesn't exist anywhere in this codebase yet and needs its own
  // story — this deliberately ships the button inert (visible, clicking does nothing) rather than
  // waiting for that story to land. `ZimmProductCardContent`'s own mirrored co-requirement
  // (`showNotifyMeButton`) changed the same way, for the same reason — see that file's own note.
  const showNotifyMe = Boolean(isUnavailable);

  // Fix C (SFCC-236 Step 6 visual audit): confirmed against the real Figma Default+Hover
  // instances for every variation × Unavailable=True (including the Mobile/no-hover variants,
  // which show the only state a touch pointer ever sees) that `isUnavailable` suppresses the
  // size-selector hover-swap entirely — there is nothing to pick for an out-of-stock item, so
  // price stays visible on hover instead of being replaced. Home's own hover-reveals-price
  // behaviour is untouched by this — Home's Unavailable Hover reference (node 40002163:39990)
  // still reveals price on hover exactly as the in-stock case does; only the PLP/Wishlist
  // hide-and-swap needs suppressing.
  //
  // Suppressing `hoverRevealsSizeSelector` alone is not enough to reproduce that reference: it
  // only stops the organism adding its OWN hide/reveal pair, which leaves the size-chip wrapper
  // at Content's own natural (visible) default — briefly shipped that way mid-fix and caught by
  // re-checking the real Storybook render against Figma, which confirms size chips stay hidden
  // in BOTH states when unavailable, not just "not swapped in on hover". `sizeSelectorUnavailable`
  // carries that permanent hide separately, unconditional (not gated by hover-capability at all —
  // this isn't a pointer-type concern, unlike everything else in this recipe).
  //
  // Follow-up (full-sweep visual audit against every variant x state node, not just the sampled
  // set the first pass checked): `sizeSelectorUnavailable` ALSO drives a mobile-only price hide
  // in the recipe now (Fix F) — confirmed via `get_metadata` that Mobile PLP/Wishlist Unavailable
  // hide price and show notify-me in its place (`40002163:41488`, `40002187:58475`), the OPPOSITE
  // of Desktop's additive treatment (price stays, notify-me is simply added on hover). Nothing
  // extra is computed here for it — the recipe reads the same `sizeSelectorUnavailable` boolean
  // for both the size-chip permanent-hide and this new price rule, since both only ever apply to
  // the same PLP/Wishlist-and-unavailable case.
  const hoverHidesPrice = config.hoverHidesPrice && !isUnavailable;
  const hoverRevealsSizeSelector = config.hoverRevealsSizeSelector && !isUnavailable;
  const sizeSelectorUnavailable = config.hoverRevealsSizeSelector && Boolean(isUnavailable);

  // Fix G (same follow-up audit): Wishlist-only. `hideBagOnMobile` alone only covers PLP's own
  // unconditional mobile hide; Wishlist's bag stays visible on mobile UNLESS unavailable
  // (`40002187:58470` vs `40002187:58473`). Combined into one boolean here rather than exposing
  // both conditions to the recipe — it only needs to know "hide this on mobile for this render",
  // not why.
  const bagHiddenOnMobile = config.hideBagOnMobile || (config.hideBagOnMobileWhenUnavailable && Boolean(isUnavailable));

  const removeLabel = config.showRemove ? intl.formatMessage(messages.removeFromWishlist, { name }) : undefined;
  const resolvedBadge = config.showBadge ? badge : undefined;

  const styles = useMultiStyleConfig(ZIMM_PRODUCT_CARD_RECIPE, {
    hoverHidesPrice,
    hoverRevealsPrice: config.hoverRevealsPrice,
    hoverRevealsSizeSelector,
    sizeSelectorUnavailable,
    // Same Fix C audit: notify-me is hover-gated on hover-capable devices (hidden by default,
    // revealed on hover/focus-visible — every Desktop Unavailable+Hover reference shows it
    // appearing only there) but unconditionally visible on touch (every Mobile Unavailable
    // reference — the only state a touch pointer ever sees, per the "Mobile has no Hover variant"
    // finding — shows it plainly, with no hover to reveal it). This flag is deliberately keyed off
    // `showNotifyMe` itself, not `variation`: the confirmed behaviour is identical across all
    // three variations, and the recipe's selector simply matches nothing when notify-me isn't
    // mounted at all.
    hoverRevealsNotifyMe: showNotifyMe,
    hideBagOnMobile: bagHiddenOnMobile,
  });

  // Fix I — see the module docblock's own "Fix I" paragraph for the full narrative. Guards the
  // two arrow buttons and the remove/close button (all real <button> descendants of
  // ZimmProductCardMedia) so paging/onRemove keep firing on their own and never also navigate.
  // Fix L (CodeRabbit finding on PR #85, verified correct before fixing): a real <a> lets the
  // browser handle Cmd/Ctrl-click (new tab), Shift-click (new window), and non-primary-button
  // clicks natively — this handler only ever replaces the in-page navigation an UNMODIFIED,
  // primary-button click used to reach via the stretched-link overlay (see Fix I). Bailing out on
  // any of those instead of forcing an in-tab `history.push` regardless of what the shopper
  // actually asked for; a no-op here is a real gap (it doesn't open a new tab the way a real link
  // would) but is strictly better than silently overriding an explicit "open elsewhere" signal.
  const handleMediaClick = (event: React.MouseEvent<HTMLDivElement>): void => {
    const target = event.target as HTMLElement;
    if (target.closest('button')) return;
    if (event.metaKey || event.ctrlKey || event.shiftKey || event.altKey || event.button !== 0) return;
    history.push(href);
  };

  return (
    <chakra.div __css={styles.root} data-testid="product-card" data-product-id={id}>
      {config.showContent ? (
        <>
          <ZimmProductCardMedia
            primaryImage={primaryImage}
            primaryImageFallbacks={primaryImageFallbacks}
            hoverImage={hoverImage}
            images={images}
            swapOnHover={config.showHoverImageSwap && Boolean(hoverImage)}
            showControls
            badge={resolvedBadge}
            onRemove={onRemove}
            removeLabel={removeLabel}
            onClick={handleMediaClick}
            priority={priority}
            widths={imageWidths}
            quality={imageQuality}
          />
          <ZimmProductCardContent
            name={name}
            headingLevel={headingLevel}
            href={href}
            currentPrice={currentPrice}
            originalPrice={originalPrice}
            isPriceLoading={isPriceLoading}
            showAttributes={config.showAttributes}
            colour={colour}
            size={size}
            showBag={config.showBag}
            isUnavailable={isUnavailable}
            onAddToBag={onAddToBag}
            showSizeChips={config.showSizeChips}
            sizeChipValues={sizeChipValues}
            selectedSizeChipValue={selectedSizeChipValue}
            onSizeChipSelect={onSizeChipSelect}
            showNotifyMe={showNotifyMe}
            onNotifyMe={onNotifyMe}
          />
        </>
      ) : (
        // recentlyViewed only (config.showContent === false): ZimmProductCardContent never
        // mounts, so its own name-link — otherwise this card's only keyboard-reachable element —
        // is replaced by this minimal Link wrapping Media directly. See the module docblock's own
        // `recentlyViewed` paragraph for the full rationale. No `onClick` on Media here: a real
        // `<Link>` containing the image needs no click-passthrough workaround at all — that
        // mechanism exists only to route around the OTHER variations' invisible stretched-overlay
        // `<a>`, which this branch has no need for since the link directly wraps the image
        // instead of sitting behind it.
        <MediaOnlyCardLink
          to={href}
          display="block"
          // Every other interactive control in this component family (zimmColourSwatch,
          // zimmNavItem, zimmSizeChip, zimmButton, Media's own close/arrowButton, Content's
          // nameLink/bagButton/notifyMeButton) carries an explicit focus-ring rule from the same
          // registered tokens (zimmTokens.ts:68-69) — this link is no exception, even though it's
          // a plain sx override rather than a recipe part (overrideStyling.md hierarchy point 4:
          // sx for a one-off gap, referencing already-registered tokens, not a new recipe).
          sx={{
            _focusVisible: {
              outline: 'var(--zimm-focus-ring)',
              outlineOffset: 'var(--zimm-focus-ring-offset-default)',
            },
          }}
        >
          <VisuallyHidden>{name}</VisuallyHidden>
          <ZimmProductCardMedia
            // alt overridden to '' on these two copies only — never on the shared
            // ZimmProductCardMediaImage type, never in the showContent branch above. The
            // VisuallyHidden name immediately above already supplies this link's accessible name;
            // without this override the image's own real alt text would be concatenated onto it
            // too (standard name-from-content computation for a link containing an <img>),
            // producing a redundant, doubled-up announcement instead of just the product name.
            // alt="" is the standard WCAG technique for exactly this case — a decorative-in-context
            // image inside an already-labelled link.
            primaryImage={{ ...primaryImage, alt: '' }}
            // Same alt='' override as primaryImage above, applied per-candidate — whichever one
            // actually ends up rendered (see ZimmProductCardMedia's own cascading-fallback
            // contract) must carry the same decorative-in-context treatment, not just the first.
            primaryImageFallbacks={primaryImageFallbacks?.map((image) => ({ ...image, alt: '' }))}
            hoverImage={hoverImage ? { ...hoverImage, alt: '' } : undefined}
            // images is deliberately NOT forwarded here (found during this step's own Tester
            // pass): ZimmProductCardMedia's gallery mode activates purely off images.length > 1,
            // independent of any VARIATION_CONFIG flag. If this variation ever received a real
            // multi-image gallery, it would nest carousel arrow <button>s inside this branch's own
            // wrapping <Link> (invalid nested-interactive HTML), let a "Next image" click bubble
            // up and also fire this Link's own PDP navigation (a real double-navigation bug), and
            // concatenate every slide's own un-blanked alt text onto the link's accessible name —
            // the same bug just fixed for primaryImage/hoverImage above, multiplied across every
            // slide. This variation's entire design intent is single-image (AC 3), so the fix is
            // simply never forwarding images here — ZimmProductCardMedia already defaults to
            // single-image mode with no images prop supplied.
            swapOnHover={config.showHoverImageSwap && Boolean(hoverImage)}
            showControls
            badge={resolvedBadge}
            priority={priority}
            widths={imageWidths}
            quality={imageQuality}
          />
        </MediaOnlyCardLink>
      )}
    </chakra.div>
  );
};

export default ZimmProductCard;
