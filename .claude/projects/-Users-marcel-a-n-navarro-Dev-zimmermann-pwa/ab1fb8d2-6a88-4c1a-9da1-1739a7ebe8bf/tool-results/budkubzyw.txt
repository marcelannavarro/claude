   478	        const prod = variant || item.product || product;
   479	        return {
   480	          productId: prod?.productId || prod?.id, // productId for variant, id for product
   481	          price: prod?.price,

   485	      // Add inventory IDs for pickup items using the hook helper
   486	      productItems = addInventoryIdsToPickupItems(productItems, pickupInStoreMap, selectedStore);
   487	      // Defensive check: This block ensures that if, for any reason, pickup is selected for a product but no store (inventoryId) is set,

   491	          // Index-type assertion, not one of the "SDK param widening" casts described above:

   495	          // `prod?.productId || prod?.id`, and a key that is absent reads as `undefined` —
   496	          // falsy, exactly as the base file's untyped index behaves.
   497	          (item) => item.inventoryId === undefined && pickupInStoreMap[(item.productId || item.id) as string],
   498	        )
   499	      ) {
   500	        showError(
   501	          formatMessage({
   502	            id: 'product_view.error.no_store_selected_for_pickup',
   503	            defaultMessage: 'No valid store or inventory found for pickup',
   504	          }),
   505	        );
   506	        return;
   507	      }
   508	
   509	      // Check if any products have pickup selected
   510	      const hasAnyPickupSelected = hasPickupItems(productSelectionValues, pickupInStoreMap, product);
   511	
   512	      const currentShippingMethodIsPickup = isPickupMethod(basket?.shipments?.[0]?.shippingMethod);
   513	      // Only perform the check if the basket exists and has at least one item
   514	      if (!multishipEnabled && basket && basket.productItems?.length > 0) {
   515	        if (hasAnyPickupSelected && !currentShippingMethodIsPickup) {
   516	          throw new Error(
   517	            formatMessage({
   518	              id: 'product_view.error.select_ship_to_address',
   519	              defaultMessage: "Select 'Ship to Address' to match the delivery method for the items in your cart.",
   520	            }),
   521	          );
   522	        }
   523	        if (!hasAnyPickupSelected && currentShippingMethodIsPickup) {
   524	          throw new Error(
   525	            formatMessage({
   526	              id: 'product_view.error.select_pickup_in_store',
   527	              defaultMessage: "Select 'Pick Up in Store' to match the delivery method for the items in your cart.",
   528	            }),
   529	          );
   530	        }
   531	      }
   532	
   533	      // Fetch and assign a suitable shipment for product items
   534	      const targetShipmentId = await getShipmentIdForItems(hasAnyPickupSelected, selectedStore);
   535	
   536	      if (targetShipmentId) {
   537	        productItems = productItems.map((item) => ({
   538	          ...item,
   539	          shipmentId: targetShipmentId,
   540	        }));
   541	      }
   542	
   543	      const basketResponse = await addItemToNewOrExistingBasket(productItems);
   544	
   545	      // Configure shipping method for default shipment based on pickup selection
   546	      await updateDefaultShipmentIfNeeded(basketResponse, targetShipmentId, hasAnyPickupSelected, selectedStore);
   547	
   548	      const productItemsForEinstein = productSelectionValues.map(({ product, variant, quantity }) => ({
   549	        product,
   550	        productId: variant?.productId || product?.id,
   551	        price: variant?.price || product?.price,
   552	        quantity,
   553	      }));
   554	      einstein.sendAddToCart(productItemsForEinstein);
   555	
   556	      return productSelectionValues;
   557	    } catch (error) {
   558	      // `catch` binds `unknown` under strict mode, and a thrown non-Error has no `.message`,
   559	      // so narrow instead of asserting. Behaviour is unchanged — the previous
   560	      // `(error as Error).message` also yielded `undefined` for a non-Error — but this was the
   561	      // only `as` in this file without the inline reason typescript.md requires.
   562	      showError(error instanceof Error ? error.message : undefined);
   563	    }
   564	  };
   565	
   566	  /**************** Product Set/Bundles Handlers ****************/
   567	  const handleChildProductValidation = useCallback(() => {
   568	    // Run validation for all child products. This will ensure the error
   569	    // messages are shown.
   570	    Object.values(childProductRefs.current).forEach(({ validateOrderability }) => {
   571	      validateOrderability({ scrollErrorIntoView: false });
   572	    });
   573	
   574	    // Using ot state for which child products are selected, scroll to the first
   575	    // one that isn't selected and requires a variant selection.
   576	    const selectedProductIds = Object.keys(childProductSelection);
   577	    const firstUnselectedProduct = product?.childProducts?.find(
   578	      ({ product: childProduct }: { product: SfccProduct }) => {
   579	        // Skip validation for standard products (no variations)
   580	        if (childProduct.type?.item) {
   581	          return false;
   582	        }
   583	        return !selectedProductIds.includes(childProduct.id);
   584	      },
   585	    )?.product;
   586	
   587	    if (firstUnselectedProduct) {
   588	      // Get the reference to the product view and scroll to it.
   589	      const { ref } = childProductRefs.current[firstUnselectedProduct.id];
   590	
   591	      // Deviation from the base file: `?.` on `ref`, required by strictNullChecks —
   592	      // `ChildProductRef.ref` is typed `HTMLDivElement | null`. The base reads
   593	      // `ref.scrollIntoView` unguarded and throws when the child view has not mounted; ours
   594	      // short-circuits and returns `false`. The unguarded call inside the block is reached
   595	      // only once this guard has passed.
   596	      if (ref?.scrollIntoView) {
   597	        ref.scrollIntoView({
   598	          behavior: 'smooth',
   599	          block: 'end',
   600	        });
   601	      }
   602	
   603	      return false;
   604	    }
   605	
   606	    return true;
   607	  }, [product, childProductSelection]);
   608	
   609	  /**************** Product Set Handlers ****************/
   610	  // 1. Gather the selected child products from state.
   611	  // 2. Call handleAddToCart with the selected products.
   612	  // 3. The add-to-cart modal will be opened in handleAddToCart.
   613	  const handleProductSetAddToCart = () => {
   614	    // Get all the selected products, and pass them to the addToCart handler which
   615	    // accepts an array.
   616	    const productSelectionValues = Object.values(childProductSelection);
   617	    return handleAddToCart(productSelectionValues);
   618	  };
   619	
   620	  /**************** Product Bundle Handlers ****************/
   621	  // Top level bundle does not have variants
   622	  const handleProductBundleAddToCart = async ([{ quantity: selectedQuantity }]: Array<{ quantity?: number }>) => {
   623	    try {
   624	      const childProductSelections = Object.values(childProductSelection);
   625	      // Check if any products have pickup selected (including main product and bundle items)
   626	      const bundleSelectionValues = [{ product, variant: null, selectedQuantity }, ...childProductSelections];
   627	      const hasAnyPickupSelected = hasPickupItems(bundleSelectionValues, pickupInStoreMap, product);
   628	
   629	      // Check for delivery method conflicts before adding to cart
   630	      if (!multishipEnabled && basket && basket.productItems?.length > 0) {
   631	        const currentShippingMethod = basket?.shipments?.[0]?.shippingMethod;
   632	        const currentShippingMethodIsPickup = isPickupMethod(currentShippingMethod);
   633	
   634	        // If there's no shipping method, treat it as non-pickup (ship to address)
   635	        if (hasAnyPickupSelected && (!currentShippingMethod || !currentShippingMethodIsPickup)) {
   636	          throw new Error(
   637	            formatMessage({
   638	              id: 'product_view.error.select_ship_to_address',
   639	              defaultMessage: "Select 'Ship to Address' to match the delivery method for the items in your cart.",
   640	            }),
   641	          );
   642	        } else if (!hasAnyPickupSelected && currentShippingMethod && currentShippingMethodIsPickup) {
   643	          throw new Error(
   644	            formatMessage({
   645	              id: 'product_view.error.select_pickup_in_store',
   646	              defaultMessage: "Select 'Pick Up in Store' to match the delivery method for the items in your cart.",
   647	            }),
   648	          );
   649	        }
   650	      }
   651	
   652	      let productItems: BasketLineItem[] = [
   653	        {
   654	          productId: product.id,
   655	          price: product.price,
   656	          quantity: selectedQuantity,
   657	          // The add item endpoint in the shopper baskets API does not respect variant selections
   658	          // for bundle children, so we have to make a follow up call to update the basket
   659	          // with the chosen variant selections
   660	          bundledProductItems: childProductSelections.map((child) => {
   661	            return {
   662	              productId: child.variant?.productId || child.product?.id,
   663	              quantity: child.quantity,
   664	            };
   665	          }),
   666	        },
   667	      ];
   668	
   669	      // Add inventory IDs for pickup items using the hook helper
   670	      productItems = addInventoryIdsToPickupItems(productItems, pickupInStoreMap, selectedStore);
   671	
   672	      // Fetch and assign a suitable shipment for product items
   673	      const targetShipmentId = await getShipmentIdForItems(hasAnyPickupSelected, selectedStore);
   674	
   675	      if (targetShipmentId) {
   676	        productItems = productItems.map((item) => ({
   677	          ...item,
   678	          shipmentId: targetShipmentId,
   679	        }));
   680	      }
   681	
   682	      const res = await addItemToNewOrExistingBasket(productItems);
   683	
   684	      const bundleChildMasterIds = childProductSelections.map((child) => {
   685	        return child.product.id;
   686	      });
   687	
   688	      // since the returned data includes all products in basket
   689	      // here we compare list of productIds in bundleProductItems of each productItem to filter out the
   690	      // current bundle that was last added into cart
   691	      // Deviation from the base file: `?.` on `productItems`. The SDK types it optional; the
   692	      // base file assumes the just-created basket always has line items.
   693	      const currentBundle = res.productItems?.find((productItem: SfccProduct) => {
   694	        if (!productItem.bundledProductItems?.length) return;
   695	        const bundleChildIds = productItem.bundledProductItems?.map((item: { productId?: string }) => {
   696	          // seek out the bundle child that still uses masterId as product id
   697	          return item.productId;
   698	        });
   699	        return bundleChildIds.every((id: string) => bundleChildMasterIds.includes(id));
   700	      });
   701	
   702	      const itemsToBeUpdated = getUpdateBundleChildArray(currentBundle, childProductSelections);
   703	
   704	      if (itemsToBeUpdated.length) {
   705	        // make a follow up call to update child variant selection for product bundle
   706	        // since add item endpoint doesn't currently consider product bundle child variants
   707	        await updateItemsInBasketMutation.mutateAsync({
   708	          // SDK param widening — `method` is a real, honoured option on the v2 baskets
   709	          // mutation (the base file relies on it to send PATCH rather than the default), but
   710	          // it is absent from the published mutation-parameter type. Dropping it would change
   711	          // the HTTP verb, so the argument is asserted instead.
   712	          method: 'PATCH',
   713	          parameters: {
   714	            // `res` is the basket this call just wrote to.
   715	            basketId: res.basketId as string,
   716	          },
   717	          body: itemsToBeUpdated,
   718	        } as Parameters<typeof updateItemsInBasketMutation.mutateAsync>[0]);
   719	      }
   720	
   721	      // Configure shipping method based on pickup selection
   722	      await updateDefaultShipmentIfNeeded(res, targetShipmentId, hasAnyPickupSelected, selectedStore);
   723	
   724	      einstein.sendAddToCart(productItems);
   725	
   726	      return childProductSelections;
   727	    } catch (error) {
   728	      showError(error);
   729	    }
   730	  };
   731	
   732	  /**************** Einstein ****************/
   733	  useEffect(() => {
   734	    if (product && product.type.set) {
   735	      einstein.sendViewProduct(product);
   736	      dataCloud.sendViewProduct(product);
   737	      const childrenProducts = product.setProducts;
   738	      childrenProducts.map((child: SfccProduct) => {
   739	        try {
   740	          einstein.sendViewProduct(child);
   741	        } catch (err) {
   742	          logger.error('Einstein sendViewProduct error', {
   743	            namespace: 'ProductDetail.useEffect',
   744	            additionalProperties: { error: err, child },
   745	          });
   746	        }
   747	        activeData.sendViewProduct(category, child, 'detail');
   748	        dataCloud.sendViewProduct(child);
   749	      });
   750	    } else if (product) {
   751	      try {
   752	        einstein.sendViewProduct(product);
   753	      } catch (err) {
   754	        logger.error('Einstein sendViewProduct error', {
   755	          namespace: 'ProductDetail.useEffect',
   756	          additionalProperties: { error: err, product },
   757	        });
   758	      }
   759	      activeData.sendViewProduct(category, product, 'detail');
   760	      dataCloud.sendViewProduct(product);
   761	    }
   762	  }, [product]);
   763	
   764	  /**************** SFCC-89 — static product information ****************/
   765	  // This formatter never executes. The page reads only `modelInfo` and `styleCode` off the mapper
   766	  // — the meta block below the accordions carries no price at all (`ZimmProductMeta`'s props are
   767	  // `modelInfo`/`styleCode`) — so `null` is passed for `priceData` below and the mapper's price
   768	  // branch (`typeof priceData?.currentPrice === 'number'`) is always false. It exists solely to
   769	  // satisfy `selectPdpProductInfo`'s required `formatPrice` parameter, which stays required so
   770	  // that module can hold its own contract of being free of react-intl.
   771	  //
   772	  // `partials/staticProductHeader.tsx` declares an identical formatter and that one IS live. The
   773	  // two cannot disagree on a rendered price, because only one of them ever formats anything.
   774	  const formatPrice = (amount: number, priceCurrency: string) =>
   775	    formatNumber(amount, { style: 'currency', currency: priceCurrency });
   776	
   777	  // Model copy and the style code are not part of the content group: the design places them at
   778	  // the foot of the column, below the accordions, so the page positions them itself.
   779	  const staticInfo = selectPdpProductInfo(product, null, product?.currency ?? '', formatPrice);
   780	
   781	  return (
   782	    <Box className="sf-product-detail-page" layerStyle="page" data-testid="product-details-page">
   783	      <Heading as="h1" srOnly>
   784	        <FormattedMessage defaultMessage="Product Details" id="product_detail.title.product_details" />
   785	      </Heading>
   786	      <Helmet>
   787	        <title>{product?.pageTitle}</title>
   788	        {product?.pageMetaTags?.length > 0 &&
   789	          product.pageMetaTags.map(({ id, value }: { id: string; value: string }) => (
   790	            <meta name={id} content={value} key={id} />
   791	          ))}
   792	        {/* Fallback for description if not included in pageMetaTags */}
   793	        {!product?.pageMetaTags?.some((tag: { id: string }) => tag.id === 'description') &&
   794	          product?.pageDescription && <meta name="description" content={product.pageDescription} />}
   795	      </Helmet>
   796	
   797	      <Stack spacing={16}>
   798	        {isProductASet || isProductABundle ? (
   799	          <Fragment>
   800	            <Island hydrateOn={'visible'}>
   801	              <ProductView
   802	                product={product}
   803	                category={primaryCategory?.parentCategoryTree || []}
   804	                addToCart={isProductASet ? handleProductSetAddToCart : handleProductBundleAddToCart}
   805	                addToWishlist={handleAddToWishlist}
   806	                isProductLoading={isProductLoading}
   807	                isBasketLoading={isBasketLoading}
   808	                isWishlistLoading={isWishlistLoading}
   809	                validateOrderability={handleChildProductValidation}
   810	                childProductOrderability={childProductOrderability}
   811	                setSelectedBundleQuantity={setSelectedBundleQuantity}
   812	                selectedBundleParentQuantity={selectedBundleQuantity}
   813	                pickupInStore={!!pickupInStoreMap[product?.id]}
   814	                setPickupInStore={(checked: boolean) => product && handlePickupInStoreChange(product.id, checked)}
   815	                onOpenStoreLocator={onOpenStoreLocator}
   816	                showDeliveryOptions={storeLocatorEnabled}
   817	              />
   818	            </Island>
   819	
   820	            <hr />
   821	
   822	            {
   823	              // Render the child products
   824	              product?.childProducts?.map(
   825	                ({ product: childProduct, quantity: childQuantity }: { product: SfccProduct; quantity: number }) => (
   826	                  <Box key={childProduct.id} data-testid="child-product">
   827	                    <ProductView
   828	                      // Do not use an arrow function as we are manipulating the functions scope.
   829	                      ref={function (this: ProductViewRefScope, ref: ChildProductRef['ref']) {
   830	                        // Assign the "set" scope of the ref, this is how we access the internal
   831	                        // validation.
   832	                        childProductRefs.current[childProduct.id] = {
   833	                          ref,
   834	                          validateOrderability: this.validateOrderability,
   835	                        };
   836	                      }}
   837	                      product={childProduct}
   838	                      isProductPartOfSet={isProductASet}
   839	                      isProductPartOfBundle={isProductABundle}
   840	                      childOfBundleQuantity={childQuantity}
   841	                      selectedBundleParentQuantity={selectedBundleQuantity}
   842	                      addToCart={
   843	                        isProductASet
   844	                          ? (productSelectionValues: ProductSelectionValue[]) => handleAddToCart(productSelectionValues)
   845	                          : null
   846	                      }
   847	                      addToWishlist={isProductASet ? handleAddToWishlist : null}
   848	                      onVariantSelected={(product: SfccProduct, variant: SfccVariant, quantity: number) => {
   849	                        if (quantity) {
   850	                          setChildProductSelection((previousState) => ({
   851	                            ...previousState,
   852	                            [product.id]: {
   853	                              product,
   854	                              variant,
   855	                              quantity: isProductABundle ? childQuantity : quantity,
   856	                            },
   857	                          }));
   858	                        } else {
   859	                          const selections = {
   860	                            ...childProductSelection,
   861	                          };
   862	                          delete selections[product.id];
   863	                          setChildProductSelection(selections);
   864	                        }
   865	                      }}
   866	                      isProductLoading={isProductLoading}
   867	                      isBasketLoading={isBasketLoading}
   868	                      isWishlistLoading={isWishlistLoading}
   869	                      setChildProductOrderability={setChildProductOrderability}
   870	                      pickupInStore={!!pickupInStoreMap[childProductSelection[childProduct?.id]?.variant?.productId]}
   871	                      setPickupInStore={(checked: boolean) =>
   872	                        childProduct &&
   873	                        handlePickupInStoreChange(childProductSelection[childProduct?.id]?.variant?.productId, checked)
   874	                      }
   875	                      onOpenStoreLocator={onOpenStoreLocator}
   876	                      showDeliveryOptions={storeLocatorEnabled && !isProductABundle}
   877	                    />
   878	                    <InformationAccordion product={childProduct} />
   879	
   880	                    <Box display={['none', 'none', 'none', 'block']}>
   881	                      <hr />
   882	                    </Box>
   883	                  </Box>
   884	                ),
   885	              )
   886	            }
   887	          </Fragment>
   888	        ) : (
   889	          <Fragment>
   890	            <Island hydrateOn={'visible'}>
   891	              <ProductView
   892	                product={product}
   893	                category={primaryCategory?.parentCategoryTree || []}
   894	                renderHeader={(headerProps) => {
   895	                  return (
   896	                    <StaticProductHeader
   897	                      {...headerProps}
   898	                      priceData={isPriceStale ? null : variantScopedPriceData ?? headerProps.priceData}
   899	                      isPriceLoading={isPriceStale}
   900	                    />
   901	                  );
   902	                }}
   903	                addToCart={handleAddToCart}
   904	                addToWishlist={handleAddToWishlist}
   905	                isProductLoading={isProductLoading}
   906	                isBasketLoading={isBasketLoading}
   907	                isWishlistLoading={isWishlistLoading}
   908	                childProductOrderability={childProductOrderability}
   909	                setChildProductOrderability={setChildProductOrderability}
   910	                setSelectedBundleQuantity={setSelectedBundleQuantity}
   911	                selectedBundleParentQuantity={selectedBundleQuantity}
   912	                pickupInStore={!!pickupInStoreMap[product?.id]}
   913	                setPickupInStore={(checked: boolean) => product && handlePickupInStoreChange(product.id, checked)}
   914	                onOpenStoreLocator={onOpenStoreLocator}
   915	                showDeliveryOptions={storeLocatorEnabled}
   916	              />
   917	              {/*
   918	               * The accordions and the meta block deliberately share one `<Box>`, so the pair is a
   919	               * single flex item of the outer `<Stack spacing={16}>` rather than two. Chakra spaces
   920	               * a Stack with `gap` rather than a child selector, and `<Island>` emits no element of
   921	               * its own while `partialHydrationEnabled` is false, so anything left unwrapped in
   922	               * here is a direct child taking a 64px push from its neighbour. Sharing one item
   923	               * lands that 64px above the pair instead of between them, which leaves
   924	               * `ZimmProductMeta`'s recipe owning both 28px hops — the one below the accordions and
   925	               * the one between its own two rows. Hence no spacing props on the Box. The molecule
   926	               * returns null when it has neither value to show.
   927	               */}
   928	              <Box>
   929	                <InformationAccordion product={product} />
   930	                <ZimmProductMeta modelInfo={staticInfo?.modelInfo} styleCode={staticInfo?.styleCode} />
   931	              </Box>
   932	            </Island>
   933	          </Fragment>
   934	        )}
   935	
   936	        {/* Product Recommendations */}
   937	        <Stack spacing={16}>
   938	          {!isProductASet && (
   939	            <Island hydrateOn={'visible'}>
   940	              <RecommendedProducts
   941	                title={
   942	                  <FormattedMessage
   943	                    defaultMessage="Complete the Set"
   944	                    id="product_detail.recommended_products.title.complete_set"
   945	                  />
   946	                }
   947	                recommender={EINSTEIN_RECOMMENDERS.PDP_COMPLETE_SET}
   948	                products={[product]}
   949	                mx={{ base: -4, md: -8, lg: 0 }}
   950	                shouldFetch={() => product?.id}
   951	              />
   952	            </Island>
   953	          )}
   954	          <Island hydrateOn={'visible'}>
   955	            <RecommendedProducts
   956	              title={
   957	                <FormattedMessage
   958	                  defaultMessage="You might also like"
   959	                  id="product_detail.recommended_products.title.might_also_like"
   960	                />
   961	              }
   962	              recommender={EINSTEIN_RECOMMENDERS.PDP_MIGHT_ALSO_LIKE}
   963	              products={[product]}
   964	              mx={{ base: -4, md: -8, lg: 0 }}
   965	              shouldFetch={() => product?.id}
   966	            />
   967	          </Island>
   968	
   969	          <Island hydrateOn={'visible'}>
   970	            <RecommendedProducts
   971	              // The Recently Viewed recommender doesn't use `products`, so instead we
   972	              // provide a key to update the recommendations on navigation.
   973	              key={location.key}
   974	              title={
   975	                <FormattedMessage
   976	                  defaultMessage="Recently Viewed"
   977	                  id="product_detail.recommended_products.title.recently_viewed"
   978	                />
   979	              }
   980	              recommender={EINSTEIN_RECOMMENDERS.PDP_RECENTLY_VIEWED}
   981	              mx={{ base: -4, md: -8, lg: 0 }}
   982	            />
   983	          </Island>
   984	        </Stack>
   985	      </Stack>
   986	    </Box>
   987	  );
   988	};
   989	
   990	ProductDetail.getTemplateName = () => 'product-detail';

   993	 * The base file declares `ProductDetail.propTypes = {match: PropTypes.object}` purely as
   994	 * documentation — the component takes no props and never reads `match`. The TypeScript
   995	 * signature above expresses the same thing at compile time, so the runtime `prop-types`
   996	 * declaration is not carried over.

   998
