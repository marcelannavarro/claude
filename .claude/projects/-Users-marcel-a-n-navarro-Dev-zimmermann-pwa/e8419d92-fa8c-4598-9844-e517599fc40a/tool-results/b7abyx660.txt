   426	  const handleAddToCart = async (productSelectionValues: ProductSelectionValue[] = []) => {
   427	    try {
   428	      let productItems: BasketLineItem[] = productSelectionValues.map((item) => {
   429	        const { variant, quantity } = item;
   430	        // Use variant if present, otherwise use the main product
   431	        const prod = variant || item.product || product;
   432	        return {
   433	          productId: prod?.productId || prod?.id, // productId for variant, id for product
   434	          price: prod?.price,
   435	          quantity,
   436	        };
   437	      });
   438	      // Add inventory IDs for pickup items using the hook helper
   439	      productItems = addInventoryIdsToPickupItems(productItems, pickupInStoreMap, selectedStore);
   440	      // Defensive check: This block ensures that if, for any reason, pickup is selected for a product but no store (inventoryId) is set,
   441	      // we show an error. With the current UI logic, this should never be reached, but it guards against unexpected state.
   442	      if (
   443	        productItems.some(
   444	          // Index-type assertion, not one of the "SDK param widening" casts described above:
   445	          // `pickupInStoreMap` is this page's own `Record<string, boolean>` state, and
   446	          // `item.productId || item.id` is `string | undefined` on `BasketLineItem`, which
   447	          // TypeScript rejects as an index (TS2538). Every item here was built above from
   448	          // `prod?.productId || prod?.id`, and a key that is absent reads as `undefined` —
   449	          // falsy, exactly as the base file's untyped index behaves.
   450	          (item) => item.inventoryId === undefined && pickupInStoreMap[(item.productId || item.id) as string],
   451	        )
   452	      ) {
   453	        showError(
   454	          formatMessage({
   455	            id: 'product_view.error.no_store_selected_for_pickup',
   456	            defaultMessage: 'No valid store or inventory found for pickup',
   457	          }),
   458	        );
   459	        return;
   460	      }
   461	
   462	      // Check if any products have pickup selected
   463	      const hasAnyPickupSelected = hasPickupItems(productSelectionValues, pickupInStoreMap, product);
   464	
   465	      const currentShippingMethodIsPickup = isPickupMethod(basket?.shipments?.[0]?.shippingMethod);
   466	      // Only perform the check if the basket exists and has at least one item
   467	      if (!multishipEnabled && basket && basket.productItems?.length > 0) {
   468	        if (hasAnyPickupSelected && !currentShippingMethodIsPickup) {
   469	          throw new Error(
   470	            formatMessage({
   471	              id: 'product_view.error.select_ship_to_address',
   472	              defaultMessage: "Select 'Ship to Address' to match the delivery method for the items in your cart.",
   473	            }),
   474	          );
   475	        }
   476	        if (!hasAnyPickupSelected && currentShippingMethodIsPickup) {
   477	          throw new Error(
   478	            formatMessage({
   479	              id: 'product_view.error.select_pickup_in_store',
   480	              defaultMessage: "Select 'Pick Up in Store' to match the delivery method for the items in your cart.",
   481	            }),
   482	          );
   483	        }
   484	      }
   485	
   486	      // Fetch and assign a suitable shipment for product items
   487	      const targetShipmentId = await getShipmentIdForItems(hasAnyPickupSelected, selectedStore);
   488	
   489	      if (targetShipmentId) {
   490	        productItems = productItems.map((item) => ({
   491	          ...item,
   492	          shipmentId: targetShipmentId,
   493	        }));
   494	      }
   495	
   496	      const basketResponse = await addItemToNewOrExistingBasket(productItems);
   497	
   498	      // Configure shipping method for default shipment based on pickup selection
   499	      await updateDefaultShipmentIfNeeded(basketResponse, targetShipmentId, hasAnyPickupSelected, selectedStore);
   500	
   501	      const productItemsForEinstein = productSelectionValues.map(({ product, variant, quantity }) => ({
   502	        product,
   503	        productId: variant?.productId || product?.id,
   504	        price: variant?.price || product?.price,
   505	        quantity,
   506	      }));
   507	      einstein.sendAddToCart(productItemsForEinstein);
   508	
   509	      return productSelectionValues;
   510	    } catch (error) {
   511	      showError((error as Error).message);
   512	    }
   513	  };

   516	  const handleChildProductValidation = useCallback(() => {
   517	    // Run validation for all child products. This will ensure the error
   518	    // messages are shown.
   519	    Object.values(childProductRefs.current).forEach(({ validateOrderability }) => {
   520	      validateOrderability({ scrollErrorIntoView: false });
   521	    });
   522	
   523	    // Using ot state for which child products are selected, scroll to the first
   524	    // one that isn't selected and requires a variant selection.
   525	    const selectedProductIds = Object.keys(childProductSelection);
   526	    const firstUnselectedProduct = product?.childProducts?.find(
   527	      ({ product: childProduct }: { product: SfccProduct }) => {
   528	        // Skip validation for standard products (no variations)
   529	        if (childProduct.type?.item) {
   530	          return false;
   531	        }
   532	        return !selectedProductIds.includes(childProduct.id);
   533	      },
   534	    )?.product;
   535	
   536	    if (firstUnselectedProduct) {
   537	      // Get the reference to the product view and scroll to it.
   538	      const { ref } = childProductRefs.current[firstUnselectedProduct.id];
   539	
   540	      // Deviation from the base file: `?.` on `ref`, required by strictNullChecks —
   541	      // `ChildProductRef.ref` is typed `HTMLDivElement | null`. The base reads
   542	      // `ref.scrollIntoView` unguarded and throws when the child view has not mounted; ours
   543	      // short-circuits and returns `false`. The unguarded call inside the block is reached
   544	      // only once this guard has passed.
   545	      if (ref?.scrollIntoView) {
   546	        ref.scrollIntoView({
   547	          behavior: 'smooth',
   548	          block: 'end',
   549	        });
   550	      }
   551	
   552	      return false;
   553	    }
   554	
   555	    return true;
   556	  }, [product, childProductSelection]);
   557	
   558	  /**************** Product Set Handlers ****************/
   559	  // 1. Gather the selected child products from state.
   560	  // 2. Call handleAddToCart with the selected products.
   561	  // 3. The add-to-cart modal will be opened in handleAddToCart.
   562	  const handleProductSetAddToCart = () => {
   563	    // Get all the selected products, and pass them to the addToCart handler which
   564	    // accepts an array.
   565	    const productSelectionValues = Object.values(childProductSelection);
   566	    return handleAddToCart(productSelectionValues);
   567	  };
   568	
   569	  /**************** Product Bundle Handlers ****************/
   570	  // Top level bundle does not have variants
   571	  const handleProductBundleAddToCart = async ([{ quantity: selectedQuantity }]: Array<{ quantity?: number }>) => {
   572	    try {
   573	      const childProductSelections = Object.values(childProductSelection);
   574	      // Check if any products have pickup selected (including main product and bundle items)
   575	      const bundleSelectionValues = [{ product, variant: null, selectedQuantity }, ...childProductSelections];
   576	      const hasAnyPickupSelected = hasPickupItems(bundleSelectionValues, pickupInStoreMap, product);
   577	
   578	      // Check for delivery method conflicts before adding to cart
   579	      if (!multishipEnabled && basket && basket.productItems?.length > 0) {
   580	        const currentShippingMethod = basket?.shipments?.[0]?.shippingMethod;
   581	        const currentShippingMethodIsPickup = isPickupMethod(currentShippingMethod);
   582	
   583	        // If there's no shipping method, treat it as non-pickup (ship to address)
   584	        if (hasAnyPickupSelected && (!currentShippingMethod || !currentShippingMethodIsPickup)) {
   585	          throw new Error(
   586	            formatMessage({
   587	              id: 'product_view.error.select_ship_to_address',
   588	              defaultMessage: "Select 'Ship to Address' to match the delivery method for the items in your cart.",
   589	            }),
   590	          );
   591	        } else if (!hasAnyPickupSelected && currentShippingMethod && currentShippingMethodIsPickup) {
   592	          throw new Error(
   593	            formatMessage({
   594	              id: 'product_view.error.select_pickup_in_store',
   595	              defaultMessage: "Select 'Pick Up in Store' to match the delivery method for the items in your cart.",
   596	            }),
   597	          );
   598	        }
   599	      }
   600	
   601	      let productItems: BasketLineItem[] = [
   602	        {
   603	          productId: product.id,
   604	          price: product.price,
   605	          quantity: selectedQuantity,
   606	          // The add item endpoint in the shopper baskets API does not respect variant selections
   607	          // for bundle children, so we have to make a follow up call to update the basket
   608	          // with the chosen variant selections
   609	          bundledProductItems: childProductSelections.map((child) => {
   610	            return {
   611	              productId: child.variant?.productId || child.product?.id,
   612	              quantity: child.quantity,
   613	            };
   614	          }),
   615	        },
   616	      ];
   617	
   618	      // Add inventory IDs for pickup items using the hook helper
   619	      productItems = addInventoryIdsToPickupItems(productItems, pickupInStoreMap, selectedStore);
   620	
   621	      // Fetch and assign a suitable shipment for product items
   622	      const targetShipmentId = await getShipmentIdForItems(hasAnyPickupSelected, selectedStore);
   623	
   624	      if (targetShipmentId) {
   625	        productItems = productItems.map((item) => ({
   626	          ...item,
   627	          shipmentId: targetShipmentId,
   628	        }));
   629	      }
   630	
   631	      const res = await addItemToNewOrExistingBasket(productItems);
   632	
   633	      const bundleChildMasterIds = childProductSelections.map((child) => {
   634	        return child.product.id;
   635	      });
   636	
   637	      // since the returned data includes all products in basket
   638	      // here we compare list of productIds in bundleProductItems of each productItem to filter out the
   639	      // current bundle that was last added into cart
   640	      // Deviation from the base file: `?.` on `productItems`. The SDK types it optional; the
   641	      // base file assumes the just-created basket always has line items.
   642	      const currentBundle = res.productItems?.find((productItem: SfccProduct) => {
   643	        if (!productItem.bundledProductItems?.length) return;
   644	        const bundleChildIds = productItem.bundledProductItems?.map((item: { productId?: string }) => {
   645	          // seek out the bundle child that still uses masterId as product id
   646	          return item.productId;
   647	        });
   648	        return bundleChildIds.every((id: string) => bundleChildMasterIds.includes(id));
   649	      });
   650	
   651	      const itemsToBeUpdated = getUpdateBundleChildArray(currentBundle, childProductSelections);
   652	
   653	      if (itemsToBeUpdated.length) {
   654	        // make a follow up call to update child variant selection for product bundle
   655	        // since add item endpoint doesn't currently consider product bundle child variants
   656	        await updateItemsInBasketMutation.mutateAsync({
   657	          // SDK param widening — `method` is a real, honoured option on the v2 baskets
   658	          // mutation (the base file relies on it to send PATCH rather than the default), but
   659	          // it is absent from the published mutation-parameter type. Dropping it would change
   660	          // the HTTP verb, so the argument is asserted instead.
   661	          method: 'PATCH',
   662	          parameters: {
   663	            // `res` is the basket this call just wrote to.
   664	            basketId: res.basketId as string,
   665	          },
   666	          body: itemsToBeUpdated,
   667	        } as Parameters<typeof updateItemsInBasketMutation.mutateAsync>[0]);
   668	      }
   669	
   670	      // Configure shipping method based on pickup selection
   671	      await updateDefaultShipmentIfNeeded(res, targetShipmentId, hasAnyPickupSelected, selectedStore);
   672	
   673	      einstein.sendAddToCart(productItems);
   674	
   675	      return childProductSelections;
   676	    } catch (error) {
   677	      showError(error);
   678	    }
   679	  };
   680	
   681	  /**************** Einstein ****************/
   682	  useEffect(() => {
   683	    if (product && product.type.set) {
   684	      einstein.sendViewProduct(product);
   685	      dataCloud.sendViewProduct(product);
   686	      const childrenProducts = product.setProducts;
   687	      childrenProducts.map((child: SfccProduct) => {
   688	        try {
   689	          einstein.sendViewProduct(child);
   690	        } catch (err) {
   691	          logger.error('Einstein sendViewProduct error', {
   692	            namespace: 'ProductDetail.useEffect',
   693	            additionalProperties: { error: err, child },
   694	          });
   695	        }
   696	        activeData.sendViewProduct(category, child, 'detail');
   697	        dataCloud.sendViewProduct(child);
   698	      });
   699	    } else if (product) {
   700	      try {
   701	        einstein.sendViewProduct(product);
   702	      } catch (err) {
   703	        logger.error('Einstein sendViewProduct error', {
   704	          namespace: 'ProductDetail.useEffect',
   705	          additionalProperties: { error: err, product },
   706	        });
   707	      }
   708	      activeData.sendViewProduct(category, product, 'detail');
   709	      dataCloud.sendViewProduct(product);
   710	    }
   711	  }, [product]);
   712	
   713	  /**************** SFCC-89 — static product information ****************/
   714	  // `partials/staticProductHeader.tsx` holds an identical copy for the header it owns. Deliberate,
   715	  // not a leftover: hoisting this into `productInfo.ts` would falsify that module's own contract
   716	  // ("Injected so this module stays free of react-intl"), and the two callers sit either side of
   717	  // the extraction boundary. Change one and change the other — otherwise the header price and the
   718	  // meta-block price can disagree on the same rendered page.
   719	  const formatPrice = (amount: number, priceCurrency: string) =>
   720	    formatNumber(amount, { style: 'currency', currency: priceCurrency });
   721	
   722	  // Model copy and the style code are not part of the content group: the design places them at
   723	  // the foot of the column, below the accordions, so the page positions them itself.
   724	  const staticInfo = selectPdpProductInfo(product, null, product?.currency ?? '', formatPrice);
   725	
   726	  return (
   727	    <Box className="sf-product-detail-page" layerStyle="page" data-testid="product-details-page">
   728	      <Heading as="h1" srOnly>
   729	        <FormattedMessage defaultMessage="Product Details" id="product_detail.title.product_details" />
   730	      </Heading>
   731	      <Helmet>
   732	        <title>{product?.pageTitle}</title>
   733	        {product?.pageMetaTags?.length > 0 &&
   734	          product.pageMetaTags.map(({ id, value }: { id: string; value: string }) => (
   735	            <meta name={id} content={value} key={id} />
   736	          ))}
   737	        {/* Fallback for description if not included in pageMetaTags */}
   738	        {!product?.pageMetaTags?.some((tag: { id: string }) => tag.id === 'description') &&
   739	          product?.pageDescription && <meta name="description" content={product.pageDescription} />}
   740	      </Helmet>
   741	
   742	      <Stack spacing={16}>
   743	        {isProductASet || isProductABundle ? (
   744	          <Fragment>
   745	            <Island hydrateOn={'visible'}>
   746	              <ProductView
   747	                product={product}
   748	                category={primaryCategory?.parentCategoryTree || []}
   749	                addToCart={isProductASet ? handleProductSetAddToCart : handleProductBundleAddToCart}
   750	                addToWishlist={handleAddToWishlist}
   751	                isProductLoading={isProductLoading}
   752	                isBasketLoading={isBasketLoading}
   753	                isWishlistLoading={isWishlistLoading}
   754	                validateOrderability={handleChildProductValidation}
   755	                childProductOrderability={childProductOrderability}
   756	                setSelectedBundleQuantity={setSelectedBundleQuantity}
   757	                selectedBundleParentQuantity={selectedBundleQuantity}
   758	                pickupInStore={!!pickupInStoreMap[product?.id]}
   759	                setPickupInStore={(checked: boolean) => product && handlePickupInStoreChange(product.id, checked)}
   760	                onOpenStoreLocator={onOpenStoreLocator}
   761	                showDeliveryOptions={storeLocatorEnabled}
   762	              />
   763	            </Island>
   764	
   765	            <hr />
   766	
   767	            {
   768	              // Render the child products
   769	              product?.childProducts?.map(
   770	                ({ product: childProduct, quantity: childQuantity }: { product: SfccProduct; quantity: number }) => (
   771	                  <Box key={childProduct.id} data-testid="child-product">
   772	                    <ProductView
   773	                      // Do not use an arrow function as we are manipulating the functions scope.
   774	                      ref={function (this: ProductViewRefScope, ref: ChildProductRef['ref']) {
   775	                        // Assign the "set" scope of the ref, this is how we access the internal
   776	                        // validation.
   777	                        childProductRefs.current[childProduct.id] = {
   778	                          ref,
   779	                          validateOrderability: this.validateOrderability,
   780	                        };
   781	                      }}
   782	                      product={childProduct}
   783	                      isProductPartOfSet={isProductASet}
   784	                      isProductPartOfBundle={isProductABundle}
   785	                      childOfBundleQuantity={childQuantity}
   786	                      selectedBundleParentQuantity={selectedBundleQuantity}
   787	                      addToCart={
   788	                        isProductASet
   789	                          ? (productSelectionValues: ProductSelectionValue[]) => handleAddToCart(productSelectionValues)
   790	                          : null
   791	                      }
   792	                      addToWishlist={isProductASet ? handleAddToWishlist : null}
   793	                      onVariantSelected={(product: SfccProduct, variant: SfccVariant, quantity: number) => {
   794	                        if (quantity) {
   795	                          setChildProductSelection((previousState) => ({
   796	                            ...previousState,
   797	                            [product.id]: {
   798	                              product,
   799	                              variant,
   800	                              quantity: isProductABundle ? childQuantity : quantity,
   801	                            },
   802	                          }));
   803	                        } else {
   804	                          const selections = {
   805	                            ...childProductSelection,
   806	                          };
   807	                          delete selections[product.id];
   808	                          setChildProductSelection(selections);
   809	                        }
   810	                      }}
   811	                      isProductLoading={isProductLoading}
   812	                      isBasketLoading={isBasketLoading}
   813	                      isWishlistLoading={isWishlistLoading}
   814	                      setChildProductOrderability={setChildProductOrderability}
   815	                      pickupInStore={!!pickupInStoreMap[childProductSelection[childProduct?.id]?.variant?.productId]}
   816	                      setPickupInStore={(checked: boolean) =>
   817	                        childProduct &&
   818	                        handlePickupInStoreChange(childProductSelection[childProduct?.id]?.variant?.productId, checked)
   819	                      }
   820	                      onOpenStoreLocator={onOpenStoreLocator}
   821	                      showDeliveryOptions={storeLocatorEnabled && !isProductABundle}
   822	                    />
   823	                    <InformationAccordion product={childProduct} />
   824	
   825	                    <Box display={['none', 'none', 'none', 'block']}>
   826	                      <hr />
   827	                    </Box>
   828	                  </Box>
   829	                ),
   830	              )
   831	            }
   832	          </Fragment>
   833	        ) : (
   834	          <Fragment>
   835	            <Island hydrateOn={'visible'}>
   836	              <ProductView
   837	                product={product}
   838	                category={primaryCategory?.parentCategoryTree || []}
   839	                renderHeader={(headerProps) => <StaticProductHeader {...headerProps} />}
   840	                addToCart={handleAddToCart}
   841	                addToWishlist={handleAddToWishlist}
   842	                isProductLoading={isProductLoading}
   843	                isBasketLoading={isBasketLoading}
   844	                isWishlistLoading={isWishlistLoading}
   845	                childProductOrderability={childProductOrderability}
   846	                setChildProductOrderability={setChildProductOrderability}
   847	                setSelectedBundleQuantity={setSelectedBundleQuantity}
   848	                selectedBundleParentQuantity={selectedBundleQuantity}
   849	                pickupInStore={!!pickupInStoreMap[product?.id]}
   850	                setPickupInStore={(checked: boolean) => product && handlePickupInStoreChange(product.id, checked)}
   851	                onOpenStoreLocator={onOpenStoreLocator}
   852	                showDeliveryOptions={storeLocatorEnabled}
   853	              />
   854	              {/*
   855	               * The accordions and the meta block deliberately share one `<Box>`, so the pair is a
   856	               * single flex item of the outer `<Stack spacing={16}>` rather than two. Chakra spaces
   857	               * a Stack with `gap` rather than a child selector, and `<Island>` emits no element of
   858	               * its own while `partialHydrationEnabled` is false, so anything left unwrapped in
   859	               * here is a direct child taking a 64px push from its neighbour. Sharing one item
   860	               * lands that 64px above the pair instead of between them, which leaves
   861	               * `ZimmProductMeta`'s recipe owning both 28px hops — the one below the accordions and
   862	               * the one between its own two rows. Hence no spacing props on the Box. The molecule
   863	               * returns null when it has neither value to show.
   864	               */}
   865	              <Box>
   866	                <InformationAccordion product={product} />
   867	                <ZimmProductMeta modelInfo={staticInfo?.modelInfo} styleCode={staticInfo?.styleCode} />
   868	              </Box>
   869	            </Island>
   870	          </Fragment>
   871	        )}
   872	
   873	        {/* Product Recommendations */}
   874	        <Stack spacing={16}>
   875	          {!isProductASet && (
   876	            <Island hydrateOn={'visible'}>
   877	              <RecommendedProducts
   878	                title={
   879	                  <FormattedMessage
   880	                    defaultMessage="Complete the Set"
   881	                    id="product_detail.recommended_products.title.complete_set"
   882	                  />
   883	                }
   884	                recommender={EINSTEIN_RECOMMENDERS.PDP_COMPLETE_SET}
   885	                products={[product]}
   886	                mx={{ base: -4, md: -8, lg: 0 }}
   887	                shouldFetch={() => product?.id}
   888	              />
   889	            </Island>
   890	          )}
   891	          <Island hydrateOn={'visible'}>
   892	            <RecommendedProducts
   893	              title={
   894	                <FormattedMessage
   895	                  defaultMessage="You might also like"
   896	                  id="product_detail.recommended_products.title.might_also_like"
   897	                />
   898	              }
   899	              recommender={EINSTEIN_RECOMMENDERS.PDP_MIGHT_ALSO_LIKE}
   900	              products={[product]}
   901	              mx={{ base: -4, md: -8, lg: 0 }}
   902	              shouldFetch={() => product?.id}
   903	            />
   904	          </Island>
   905	
   906	          <Island hydrateOn={'visible'}>
   907	            <RecommendedProducts
   908	              // The Recently Viewed recommender doesn't use `products`, so instead we
   909	              // provide a key to update the recommendations on navigation.
   910	              key={location.key}
   911	              title={
   912	                <FormattedMessage
   913	                  defaultMessage="Recently Viewed"
   914	                  id="product_detail.recommended_products.title.recently_viewed"
   915	                />
   916	              }
   917	              recommender={EINSTEIN_RECOMMENDERS.PDP_RECENTLY_VIEWED}
   918	              mx={{ base: -4, md: -8, lg: 0 }}
   919	            />
   920	          </Island>
   921	        </Stack>
   922	      </Stack>
   923	    </Box>
   924	  );
   925	};
   926	
   927	ProductDetail.getTemplateName = () => 'product-detail';
   928

   930	 * The base file declares `ProductDetail.propTypes = {match: PropTypes.object}` purely as
   931	 * documentation — the component takes no props and never reads `match`. The TypeScript
   932	 * signature above expresses the same thing at compile time, so the runtime `prop-types`
   933	 * declaration is not carried over.
   934	 */
   935	
   936	export default ProductDetail;
